"""Guarded fixed-matrix driver; only called by the private controller."""

from __future__ import annotations

import functools
import hashlib
import importlib.util
import json
import os
from pathlib import Path
import sys


def load(name, path):
    spec = importlib.util.spec_from_file_location(name, path)
    module = importlib.util.module_from_spec(spec)
    spec.loader.exec_module(module)
    return module


def main():
    job_path = Path(sys.argv[1])
    job = json.loads(job_path.read_text())
    root = job_path.parent
    for name, expected in job["inputs"].items():
        if hashlib.sha256(Path(name).read_bytes()).hexdigest() != expected:
            raise RuntimeError("driver input hash differs")
    import sitecustomize

    if Path(sitecustomize.__file__).resolve() != root / "guard/sitecustomize.py":
        raise RuntimeError("unexpected guard origin")
    sys.path.extend([job["support"], job["psutil"], str(Path(__file__).parent)])
    import owned_harness

    measure = load("stress_measure", Path(job["measure"]))
    measure.run_harness_once = functools.partial(
        owned_harness.run_harness_once, measure
    )
    roots = {key: Path(value) for key, value in job["roots"].items()}
    guard_path = os.environ["PYTHONPATH"]
    dsn = measure.guarded_dsn(os.environ["LOCALOS_READINESS_JOURNEY_DATABASE_URL"])
    output = {
        "profile": {"warmup": 0, "serial": 1, "stress": 20, "concurrency": 2},
        "refs": job["refs"],
        "runs": {},
        "status": "started",
    }
    output["origins"] = {
        key: measure.import_origins(path, dsn, guard_path)
        for key, path in roots.items()
    }
    run_dir = root / "runs"
    run_dir.mkdir()
    try:
        serial = measure.run_interleaved(
            roots, dsn, guard_path, run_dir, job["refs"], "serial", 1
        )
        for key, source in roots.items():
            output["runs"][key] = serial[key]
            output["runs"][key].extend(
                measure.run_phase(
                    source,
                    dsn,
                    guard_path,
                    run_dir,
                    job["refs"][key],
                    "full_harness_stress",
                    20,
                    2,
                )
            )
        output["status"] = (
            "passed"
            if all(run["valid"] for rows in output["runs"].values() for run in rows)
            else "failed"
        )
    except BaseException:
        output["status"] = "failed"
        output["error_type"] = type(sys.exception()).__name__
        raise
    finally:
        (root / "driver-result.json").write_text(json.dumps(output, indent=2) + "\n")
    return 0 if output["status"] == "passed" else 1


if __name__ == "__main__":
    raise SystemExit(main())
