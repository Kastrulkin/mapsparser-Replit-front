"""Pure matrix/archive/policy contracts: never starts a process or database."""

import copy
import hashlib
from pathlib import Path
import tarfile

import pytest
import stress_controller


def matrix():
    inv = [
        ("service_compression", "apply_invariants"),
        ("operator", "confirm_invariants"),
        ("operator", "foreign_plan_untouched"),
    ]
    rows = [{"kind": "request", "success": True} for _ in range(15)]
    rows.extend(
        {"kind": "invariant", "journey": journey, "step": step, "success": True}
        for journey, step in inv
    )
    payload = {
        "status": "passed",
        "refs": stress_controller.REFS,
        "profile": {"warmup": 0, "serial": 1, "stress": 20, "concurrency": 2},
        "runs": {},
    }
    for label, ref in stress_controller.REFS.items():
        payload["runs"][label] = [
            {
                "ref": ref,
                "phase": phase,
                "sample_number": index,
                "valid": True,
                "invalid_reasons": [],
                "samples": copy.deepcopy(rows),
                "planned_database_name": label + phase + str(index),
            }
            for phase, count in [("serial", 1), ("full_harness_stress", 20)]
            for index in range(count)
        ]
    return payload


def test_exact_matrix():
    counts = stress_controller.assert_matrix(matrix())
    assert (counts["runs"], counts["requests"], counts["invariants"]) == (42, 630, 126)


@pytest.mark.parametrize(
    "change",
    [
        "missing_run",
        "duplicate_sample",
        "duplicate_database",
        "duplicate_invariant",
        "failed_request",
        "wrong_ref",
        "wrong_concurrency",
    ],
)
def test_reject_bad_matrix(change):
    payload = matrix()
    first = payload["runs"]["baseline"][0]
    if change == "missing_run":
        payload["runs"]["baseline"].pop()
    if change == "duplicate_sample":
        payload["runs"]["baseline"][-1]["sample_number"] = 0
    if change == "duplicate_database":
        payload["runs"]["baseline"][-1]["planned_database_name"] = first[
            "planned_database_name"
        ]
    if change == "duplicate_invariant":
        first["samples"][-1] = first["samples"][-2]
    if change == "failed_request":
        first["samples"][0]["success"] = False
    if change == "wrong_ref":
        first["ref"] = "other"
    if change == "wrong_concurrency":
        payload["profile"]["concurrency"] = 4
    with pytest.raises(RuntimeError):
        stress_controller.assert_matrix(payload)


@pytest.mark.parametrize(
    "name,kind",
    [
        ("/absolute", tarfile.REGTYPE),
        ("../escape", tarfile.REGTYPE),
        ("link", tarfile.SYMTYPE),
        ("hardlink", tarfile.LNKTYPE),
        ("fifo", tarfile.FIFOTYPE),
    ],
)
def test_reject_unsafe_archive(name, kind):
    member = tarfile.TarInfo(name)
    member.type = kind
    with pytest.raises(RuntimeError):
        stress_controller.safe_member(member)


def test_accept_regular_archive_member():
    stress_controller.safe_member(tarfile.TarInfo("src/module.py"))


def test_policy_protects_source_and_user_files():
    value = stress_controller.policy(Path("/private/tmp/example"), 40000)
    assert "(deny network*)" in value
    assert '(deny file-read* (subpath "/Users"))' in value
    assert '(deny file-write* (subpath "/private/tmp/example/baseline"))' in value
    assert '(deny file-write* (subpath "/private/tmp/example/current"))' in value


def test_manifest_bytes_read_exactly_once():
    class ChangingManifest:
        calls = 0

        def read_bytes(self):
            self.calls += 1
            return b'{"a":"b"}' if self.calls == 1 else b"{}"

    path = ChangingManifest()
    stress_controller.verify_reviewed_inputs(
        path, hashlib.sha256(b'{"a":"b"}').hexdigest(), {"a": "b"}
    )
    assert path.calls == 1


@pytest.mark.parametrize("case", ["missing", "wrong_hash", "wrong_content"])
def test_manifest_negative_gates(case, tmp_path):
    path = tmp_path / "reviewed.json"
    path.write_bytes(b"{}")
    expected = hashlib.sha256(b"{}").hexdigest()
    if case == "missing":
        path = None
    if case == "wrong_hash":
        expected = "wrong"
    contents = {"a": "b"} if case == "wrong_content" else {}
    with pytest.raises(RuntimeError):
        stress_controller.verify_reviewed_inputs(path, expected, contents)


def clean_finalization():
    return {
        "database_catalog": ["postgres", "template0", "template1"],
        "identity_verified_before_stop": True,
        "cluster_stop": "stopped",
        "process_cleanup": {
            "status": "clean",
            "signalled": [],
            "remaining": [],
            "errors": [],
        },
    }


def test_full_finalization_required():
    assert stress_controller.finalized_cleanly(clean_finalization())
    assert not stress_controller.finalized_cleanly(
        {
            "cluster_stop": "unexpected_exit",
            "process_cleanup": {
                "status": "clean",
                "signalled": [],
                "remaining": [],
                "errors": [],
            },
        }
    )


@pytest.mark.parametrize(
    "missing",
    [
        "database_catalog",
        "identity_verified_before_stop",
        "cluster_stop",
        "process_cleanup",
    ],
)
def test_missing_finalization_gate_fails(missing):
    result = clean_finalization()
    result.pop(missing)
    assert not stress_controller.finalized_cleanly(result)
