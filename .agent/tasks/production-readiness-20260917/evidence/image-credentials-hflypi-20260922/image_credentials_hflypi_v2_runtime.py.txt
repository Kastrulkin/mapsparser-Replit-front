"""Scan every regular byte of one pinned OCI image with redacted Gitleaks.

The scanner streams all historical layers and the OCI config without extracting
them to disk or running the image. It retains only redacted rule/line/path-hash
metadata. This is evidence for default Gitleaks rules, not proof that every
credential format or build-context artifact is absent.
"""

import argparse
import gzip
import hashlib
import json
import os
from pathlib import Path, PurePosixPath
import re
import runpy
import signal
import shutil
import subprocess
import sys
import tarfile
import threading
import time

PRIVATE = runpy.run_path(str(Path(__file__).with_name("image_private_layers_hflypi.py")))
BLOB = PRIVATE["BLOB"]
CHUNK = PRIVATE["CHUNK"]
DOCKER = PRIVATE["DOCKER"]
ENV = PRIVATE["ENV"]
IMAGE = PRIVATE["IMAGE"]
MAX_ARCHIVE = PRIVATE["MAX_ARCHIVE"]
MAX_JSON = PRIVATE["MAX_JSON"]
MAX_LAYER = PRIVATE["MAX_LAYER"]
MAX_TOTAL = PRIVATE["MAX_TOTAL"]
Meter = PRIVATE["Meter"]
Prefix = PRIVATE["Prefix"]
drain = PRIVATE["drain"]
normalized = PRIVATE["normalized"]
require = PRIVATE["require"]
small_bytes = PRIVATE["small_bytes"]
validate_graph = PRIVATE["validate_graph"]


EVIDENCE = Path("/private/tmp/localos-readiness-20260921.hfLYPi/native/evidence")
GITLEAKS = ["/usr/local/bin/gitleaks"]
GITLEAKS_FLAGS = ["stdin", "--redact=100", "--ignore-gitleaks-allow", "--gitleaks-ignore-path", "/dev/null", "--report-format", "json", "--report-path", "-", "--no-banner", "--no-color", "--timeout", "600"]
MAX_GITLEAKS_OUTPUT = 2 * 1024**2
GITLEAKS_TIMEOUT = 630
EXPORT_TIMEOUT = 1200


class ScanFailure(RuntimeError):
    """A scanner failure with safe, content-free progress diagnostics."""

    def __init__(self, message, diagnostics):
        super().__init__(message)
        self.diagnostics = diagnostics


def path_hash(value):
    return hashlib.sha256(value.encode()).hexdigest()


def kill_process_group(process):
    if process.poll() is None:
        os.killpg(process.pid, signal.SIGKILL)


def gitleaks_stream(write, ranges, progress):
    process = subprocess.Popen(GITLEAKS + GITLEAKS_FLAGS, env=ENV, stdin=subprocess.PIPE, stdout=subprocess.PIPE, stderr=subprocess.DEVNULL, start_new_session=True)
    expired = threading.Event()
    overflow = threading.Event()
    output = bytearray()
    progress["gitleaks"] = {"bytes_fed": 0, "returncode": None, "watchdog_expired": False, "output_overflow": False, "internal_timeout_seconds": 600, "watchdog_timeout_seconds": GITLEAKS_TIMEOUT, "reaped_after_input_close": False}

    def timeout():
        expired.set()
        kill_process_group(process)

    def collect():
        while True:
            block = process.stdout.read(8192)
            if not block:
                return
            if len(output) + len(block) > MAX_GITLEAKS_OUTPUT:
                overflow.set()
                kill_process_group(process)
                return
            output.extend(block)

    timer = threading.Timer(GITLEAKS_TIMEOUT, timeout)
    reader = threading.Thread(target=collect, daemon=True)
    timer.start()
    reader.start()
    try:
        require(process.stdin is not None, "gitleaks stdin unavailable")
        def feed(value):
            view = memoryview(value)
            while view:
                written = process.stdin.write(view)
                require(isinstance(written, int) and written > 0, "gitleaks stdin write failed")
                progress["gitleaks"]["bytes_fed"] += written
                view = view[written:]

        try:
            write(feed)
            process.stdin.close()
        except BrokenPipeError:
            try:
                returncode = process.wait(timeout=10)
            except subprocess.TimeoutExpired:
                kill_process_group(process)
                returncode = process.wait(timeout=10)
                progress["gitleaks"]["reaped_after_input_close"] = True
            progress["gitleaks"].update({"returncode": returncode, "watchdog_expired": expired.is_set(), "output_overflow": overflow.is_set()})
            raise ScanFailure("gitleaks input pipe closed", progress)
        returncode = process.wait(timeout=10)
        reader.join(timeout=10)
        require(not reader.is_alive(), "gitleaks output collector did not finish")
        progress["gitleaks"].update({"returncode": returncode, "watchdog_expired": expired.is_set(), "output_overflow": overflow.is_set()})
        if expired.is_set() or overflow.is_set():
            raise ScanFailure("gitleaks timed out or exceeded redacted output limit", progress)
        require(returncode in {0, 1}, "gitleaks scan failed")
        raw = json.loads(bytes(output))
        require(isinstance(raw, list), "gitleaks report is not a list")
        findings = []
        for row in raw:
            require(isinstance(row, dict) and isinstance(row.get("Match"), str) and "REDACTED" in row["Match"] and row.get("Secret") == "REDACTED", "gitleaks output was not fully redacted")
            line = row.get("StartLine")
            rule = row.get("RuleID")
            require(isinstance(line, int) and line > 0 and isinstance(rule, str) and re.fullmatch(r"[A-Za-z0-9_.-]{1,128}", rule) is not None, "gitleaks finding metadata is invalid")
            mapped = next((item for item in ranges if item[0] <= line <= item[1]), None)
            require(mapped is not None, "gitleaks finding does not map to a scanned member")
            findings.append({"path_sha256": mapped[2], "line": line - mapped[0] + 1, "rule_id": rule})
        require((returncode == 1) == bool(findings), "gitleaks exit/result disagreement")
        return findings
    finally:
        timer.cancel()
        kill_process_group(process)
        try:
            process.wait(timeout=10)
        except subprocess.TimeoutExpired:
            raise RuntimeError("gitleaks process cleanup failed")


def scan_layer(source, layer_index):
    layer = Meter(source, MAX_LAYER)
    archive = tarfile.open(fileobj=layer, mode="r|")
    counts = {"entries": 0, "regular_files": 0, "regular_bytes": 0, "links": 0, "whiteouts": 0}
    ranges = []
    line = 1
    progress = {"layer_index": layer_index, "completed_members": 0, "entries_seen": 0, "regular_bytes_completed": 0}

    def write(destination):
        nonlocal line
        for entry in archive:
            counts["entries"] += 1
            progress["entries_seen"] = counts["entries"]
            require(counts["entries"] <= 500000, "layer member limit exceeded")
            name = normalized(entry.name)
            if PurePosixPath(name).name.startswith(".wh."):
                counts["whiteouts"] += 1
            if entry.isreg():
                require(entry.sparse is None and 0 <= entry.size <= 1024**3, "unsupported regular layer member")
                member = archive.extractfile(entry)
                require(member is not None, "regular member content unavailable")
                marker = b"\n__HFLYPI_MEMBER=" + path_hash(name).encode() + b"\n"
                destination(marker)
                line += marker.count(b"\n")
                start = line
                remaining = entry.size
                member_lines = 1
                while remaining:
                    chunk = member.read(min(CHUNK, remaining))
                    require(chunk, "regular member truncated")
                    destination(chunk)
                    remaining -= len(chunk)
                    member_lines += chunk.count(b"\n")
                member.close()
                ranges.append((start, start + member_lines - 1, path_hash(name)))
                destination(b"\n")
                line = start + member_lines
                counts["regular_files"] += 1
                counts["regular_bytes"] += entry.size
                progress["completed_members"] = counts["regular_files"]
                progress["regular_bytes_completed"] = counts["regular_bytes"]
            elif entry.issym() or entry.islnk():
                counts["links"] += 1
            else:
                require(entry.isdir() or entry.isdev() or entry.isfifo(), "unsupported layer member kind")

    try:
        findings = gitleaks_stream(write, ranges, progress)
    finally:
        archive.close()
    drain(layer)
    return {**counts, "uncompressed_bytes": layer.count, "diff_id": "sha256:" + layer.digest.hexdigest(), "credential_findings": findings}


def scan_config(config, progress):
    value = json.dumps(config, sort_keys=True, separators=(",", ":")).encode()
    digest = path_hash("oci-config")
    progress.update({"layer_index": None, "completed_members": 1, "entries_seen": 1, "regular_bytes_completed": len(value)})
    return gitleaks_stream(lambda destination: destination(value), [(1, value.count(b"\n") + 1, digest)], progress)


def scan_archive(source, expected_diff_ids):
    measured = Meter(source, MAX_ARCHIVE)
    archive = tarfile.open(fileobj=measured, mode="r|")
    seen, metadata, layers, sizes = set(), {}, {}, {}
    completed_layer_progress = []
    layer_index = 0
    total = 0
    try:
        for entry in archive:
            name = normalized(entry.name)
            require(name not in seen and len(seen) < 100, "duplicate or excessive OCI members")
            seen.add(name)
            if entry.isdir():
                require(name in {"blobs", "blobs/sha256"}, "unexpected OCI directory")
                continue
            require(entry.isreg() and entry.sparse is None, "nonregular OCI member")
            sizes[name] = entry.size
            match = BLOB.fullmatch(name)
            require(match is not None or name in {"index.json", "manifest.json", "oci-layout"}, "unexpected OCI member")
            content = Meter(archive.extractfile(entry), min(MAX_ARCHIVE, entry.size))
            prefix = content.read(2)
            if prefix == b"\x1f\x8b":
                require(match is not None, "nonblob gzip member")
                compressed = gzip.GzipFile(fileobj=Prefix(prefix, content), mode="rb")
                try:
                    try:
                        row = scan_layer(compressed, layer_index)
                    except ScanFailure:
                        error = sys.exc_info()[1]
                        error.diagnostics["completed_layers"] = completed_layer_progress
                        raise
                finally:
                    compressed.close()
                drain(content)
                total += row["uncompressed_bytes"]
                require(total <= MAX_TOTAL, "total expanded image exceeds limit")
                layers[name] = {"compressed_bytes": content.count, **row}
                completed_layer_progress.append({"layer_index": layer_index, "entries_seen": row["entries"], "completed_members": row["regular_files"], "regular_bytes_completed": row["regular_bytes"], "uncompressed_bytes": row["uncompressed_bytes"]})
                layer_index += 1
            else:
                require(entry.size <= MAX_JSON, "unknown large non-gzip blob")
                metadata[name] = json.loads(prefix + small_bytes(content))
            require(content.count == entry.size, "OCI member truncated")
            if match is not None:
                require(content.digest.hexdigest() == match.group(1), "OCI content digest mismatch")
    finally:
        archive.close()
    drain(measured)
    ordered = validate_graph(metadata, layers, expected_diff_ids, sizes)
    config_path = "blobs/" + metadata["blobs/" + metadata["index.json"]["manifests"][0]["digest"].replace(":", "/")]["config"]["digest"].replace(":", "/")
    config_progress = {}
    try:
        config_findings = scan_config(metadata[config_path], config_progress)
    except ScanFailure:
        error = sys.exc_info()[1]
        error.diagnostics["completed_layers"] = completed_layer_progress
        error.diagnostics["config_progress"] = config_progress
        raise
    return {"archive_bytes": measured.count, "archive_sha256": measured.digest.hexdigest(), "expanded_bytes": total, "layers": ordered, "config_credential_findings": config_findings, "finding_count": len(config_findings) + sum(len(row["credential_findings"]) for row in ordered)}


def inspect_image():
    result = subprocess.run(DOCKER + ["image", "inspect", IMAGE], env=ENV, capture_output=True, timeout=30, check=True)
    require(len(result.stdout) <= MAX_JSON, "inspect output exceeds limit")
    rows = json.loads(result.stdout)
    require(len(rows) == 1 and rows[0]["Id"] == IMAGE and rows[0]["Os"] == "linux" and rows[0]["Architecture"] == "arm64", "image identity mismatch")
    ids = rows[0]["RootFS"]["Layers"]
    require(len(ids) == 20 and len(set(ids)) == 20, "unexpected inspected layer list")
    return ids


def write_exclusive(path, value):
    descriptor = os.open(path, os.O_WRONLY | os.O_CREAT | os.O_EXCL, 0o600)
    try:
        payload = (json.dumps(value, indent=2, sort_keys=True) + "\n").encode()
        while payload:
            payload = payload[os.write(descriptor, payload):]
    finally:
        os.close(descriptor)


def main():
    parser = argparse.ArgumentParser(description=__doc__)
    parser.add_argument("--attempt", required=True)
    args = parser.parse_args()
    require(re.fullmatch(r"v[1-9][0-9]*", args.attempt) is not None, "invalid attempt")
    require(EVIDENCE.is_dir() and not EVIDENCE.is_symlink() and EVIDENCE.resolve(strict=True) == EVIDENCE, "noncanonical evidence directory")
    require(shutil.disk_usage(EVIDENCE).free >= 5 * 1024**3, "image stream requires 5 GiB start floor")
    destination = EVIDENCE / ("image-credentials-" + args.attempt + ".json")
    require(not destination.exists() and not destination.is_symlink(), "evidence already exists")
    result = {"image": IMAGE, "attempt": args.attempt, "scope": "all-historical-layers-and-oci-config-default-gitleaks-rules", "scan_configuration": {"executable": GITLEAKS[0], "flags": GITLEAKS_FLAGS, "history": "all OCI layers plus config", "regular_file_bytes": "streamed including binary bytes; default rules may not classify arbitrary binary encodings"}}
    process = None
    timer = None
    expired = threading.Event()
    started = time.monotonic()
    try:
        ids = inspect_image()
        process = subprocess.Popen(DOCKER + ["image", "save", "--platform", "linux/arm64", IMAGE], env=ENV, stdout=subprocess.PIPE, stderr=subprocess.DEVNULL, start_new_session=True)
        timer = threading.Timer(EXPORT_TIMEOUT, lambda: (expired.set(), kill_process_group(process)))
        timer.start()
        result.update(scan_archive(process.stdout, ids))
        require(process.wait(timeout=10) == 0 and not expired.is_set(), "image export failed or timed out")
        require(inspect_image() == ids, "image identity changed during scan")
        result["status"] = "passed" if result["finding_count"] == 0 else "findings_require_review"
    except ScanFailure:
        error = sys.exc_info()[1]
        result["status"] = "failed"
        result["error_type"] = type(error).__name__
        result["failure_diagnostics"] = error.diagnostics
    except BaseException:
        result["status"] = "failed"
        result["error_type"] = type(sys.exception()).__name__
    finally:
        if timer is not None:
            timer.cancel()
        if process is not None:
            kill_process_group(process)
            try:
                process.wait(timeout=10)
            except subprocess.TimeoutExpired:
                result["status"] = "failed"
                result["cleanup_error"] = "image export process did not exit"
        result["duration_seconds"] = round(time.monotonic() - started, 3)
        result["free_bytes_after"] = shutil.disk_usage(EVIDENCE).free
        result["controller_sha256"] = hashlib.sha256(Path(__file__).read_bytes()).hexdigest()
        write_exclusive(destination, result)
    return 0 if result["status"] == "passed" else 1


if __name__ == "__main__":
    raise SystemExit(main())
