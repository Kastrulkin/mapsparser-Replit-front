"""Synthetic pure OCI credential-scanner controls; no Docker export."""

import gzip
import hashlib
import io
import json
from pathlib import Path
import runpy
import sys
import tarfile


MODULE = runpy.run_path(str(Path(__file__).with_name("image_credentials_hflypi.py")))
SCAN = MODULE["scan_archive"]
SCAN_FAILURE = MODULE["ScanFailure"]


def encoded(value):
    return json.dumps(value, sort_keys=True, separators=(",", ":")).encode()


def tar_bytes(members):
    output = io.BytesIO()
    with tarfile.open(fileobj=output, mode="w") as archive:
        for name, content in members:
            entry = tarfile.TarInfo(name)
            entry.size = len(content)
            archive.addfile(entry, io.BytesIO(content))
    return output.getvalue()


def descriptor(content, media):
    return {"mediaType": media, "digest": "sha256:" + hashlib.sha256(content).hexdigest(), "size": len(content)}


def fixture(layer_members, config_extra=None):
    blobs, layers, ids = {}, [], []
    for members in layer_members:
        raw = tar_bytes(members)
        compressed = gzip.compress(raw, mtime=0)
        item = descriptor(compressed, "application/vnd.oci.image.layer.v1.tar+gzip")
        blobs["blobs/" + item["digest"].replace(":", "/")] = compressed
        layers.append(item)
        ids.append("sha256:" + hashlib.sha256(raw).hexdigest())
    config = {"architecture": "arm64", "os": "linux", "rootfs": {"type": "layers", "diff_ids": ids}}
    if config_extra:
        config.update(config_extra)
    config_bytes = encoded(config)
    config_item = descriptor(config_bytes, "application/vnd.oci.image.config.v1+json")
    config_path = "blobs/" + config_item["digest"].replace(":", "/")
    blobs[config_path] = config_bytes
    manifest = encoded({"schemaVersion": 2, "config": config_item, "layers": layers})
    manifest_item = descriptor(manifest, "application/vnd.oci.image.manifest.v1+json")
    blobs["blobs/" + manifest_item["digest"].replace(":", "/")] = manifest
    rows = list(sorted(blobs.items())) + [
        ("index.json", encoded({"schemaVersion": 2, "manifests": [{**manifest_item, "platform": {"architecture": "arm64", "os": "linux"}}]})),
        ("manifest.json", encoded([{ "Config": config_path, "Layers": ["blobs/" + item["digest"].replace(":", "/") for item in layers]}])),
        ("oci-layout", encoded({"imageLayoutVersion": "1.0.0"})),
    ]
    return tar_bytes(rows), ids


def rejected(payload, ids):
    try:
        SCAN(io.BytesIO(payload), ids)
    except (RuntimeError, ValueError, OSError):
        return
    raise AssertionError("credential scanner falsely passed a failed scan")


def timeout_rejected(payload, ids):
    try:
        SCAN(io.BytesIO(payload), ids)
    except SCAN_FAILURE:
        error = sys.exc_info()[1]
        diagnostics = error.diagnostics
        gitleaks = diagnostics["gitleaks"]
        assert gitleaks["watchdog_expired"] is True
        assert gitleaks["output_overflow"] is False
        assert isinstance(gitleaks["bytes_fed"], int) and gitleaks["bytes_fed"] > 0
        return
    raise AssertionError("credential scanner did not reject its own watchdog timeout")


def main():
    marker = b'api_key = "AKIA1234567890ABCDEF"\n'
    globals_ = SCAN.__globals__
    assert globals_ is not MODULE
    clean, ids = fixture([[('app/main.py', b'print(1)\n')]])
    assert SCAN(io.BytesIO(clean), ids)["finding_count"] == 0
    found, ids = fixture([[('app/private/renamed.bin', marker)]])
    result = SCAN(io.BytesIO(found), ids)
    assert result["finding_count"] == 1
    finding = result["layers"][0]["credential_findings"][0]
    assert finding["path_sha256"] == hashlib.sha256(b"app/private/renamed.bin").hexdigest()
    assert set(finding) == {"path_sha256", "line", "rule_id"}
    mapped, ids = fixture([[("app/first.txt", b"first\n"), ("app/second.txt", marker.rstrip(b"\n")), ("app/third.txt", b"line\n" + marker)]])
    result = SCAN(io.BytesIO(mapped), ids)
    findings = {item["path_sha256"]: item["line"] for item in result["layers"][0]["credential_findings"]}
    assert findings[hashlib.sha256(b"app/second.txt").hexdigest()] == 1
    assert findings[hashlib.sha256(b"app/third.txt").hexdigest()] == 2
    removed, ids = fixture([[('app/old.txt', marker)], [('app/.wh.old.txt', b'')]])
    assert SCAN(io.BytesIO(removed), ids)["finding_count"] == 1
    config, ids = fixture([[('app/main.py', b'ok\n')]], {"config": {"api_key": "AKIA1234567890ABCDEF"}})
    assert len(SCAN(io.BytesIO(config), ids)["config_credential_findings"]) == 1
    original_chunk = globals_["CHUNK"]
    globals_["CHUNK"] = 8
    try:
        chunked, ids = fixture([[("app/chunk.txt", b"x" * (1024**2) + b"\x00\xff" * 4096 + marker)]])
        result = SCAN(io.BytesIO(chunked), ids)
        assert result["finding_count"] == 1
        finding = result["layers"][0]["credential_findings"][0]
        assert finding["path_sha256"] == hashlib.sha256(b"app/chunk.txt").hexdigest()
        assert finding["line"] == 1
    finally:
        globals_["CHUNK"] = original_chunk
    original = globals_["GITLEAKS"]
    globals_["GITLEAKS"] = [sys.executable, "-c", "import time; time.sleep(2)"]
    original_timeout = globals_["GITLEAKS_TIMEOUT"]
    globals_["GITLEAKS_TIMEOUT"] = 0.01
    try:
        timeout_rejected(clean, ids)
    finally:
        globals_["GITLEAKS"] = original
        globals_["GITLEAKS_TIMEOUT"] = original_timeout
    print("OCI credential controls: clean, renamed, deleted lower layer, config, chunk boundary, failure timeout and redacted mapping PASS")


if __name__ == "__main__":
    main()
