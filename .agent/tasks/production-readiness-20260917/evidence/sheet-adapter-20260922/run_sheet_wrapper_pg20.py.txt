def sha256_path(path: Path) -> str:
    digest = hashlib.sha256()
    with path.open("rb") as source:
        for block in iter(lambda: source.read(1024 * 1024), b""):
            digest.update(block)
    return digest.hexdigest()

def source_manifest(root: Path) -> dict[str, object]:
    records: list[dict[str, object]] = []
    for path in sorted(root.rglob("*")):
        relative = path.relative_to(root)
        if path.is_symlink():
            raise RuntimeError("immutable source contains a symlink")
        if path.is_file():
            records.append({"path": str(relative), "mode": path.stat().st_mode & 0o777, "sha256": sha256_path(path)})
    encoded = json.dumps(records, separators=(",", ":"), sort_keys=True).encode()
    return {"count": len(records), "sha256": hashlib.sha256(encoded).hexdigest()}

            raise RuntimeError("required bridge path is absent")
        frozen_hash = sha256_path(frozen)
        current_hash = sha256_path(current)
        if frozen_hash != current_hash:
            raise RuntimeError("frozen source does not bridge current HEAD")
        bridge.append({"path": relative, "sha256": frozen_hash})

    for private_path, workspace_path, expected_hash in (
        (PRIVATE_RUNTIME_TEST, WORKSPACE_RUNTIME_TEST, EXPECTED_RUNTIME_TEST_SHA256),

        if workspace_path.is_symlink() or not workspace_path.is_file() or sha256_path(workspace_path) != expected_hash:
            raise RuntimeError("workspace test does not match the reviewed private copy")
        bridge.append({"path": str(private_path), "sha256": expected_hash, "source_kind": "private_test_copy", "workspace_source": str(workspace_path)})

        ["git", "show", "HEAD:" + EXECUTOR_PATH], cwd=WORKSPACE,

        raise RuntimeError("frozen executor does not match committed HEAD")

        raise RuntimeError("candidate executor overlay hash changed")

def load_module(name: str, path: Path) -> object:
    if str(PSUTIL_SUPPORT) not in sys.path:
        sys.path.insert(0, str(PSUTIL_SUPPORT))
    if str(path.parent) not in sys.path:
        sys.path.insert(0, str(path.parent))
    specification = importlib.util.spec_from_file_location(name, path)
    if specification is None or specification.loader is None:
        raise RuntimeError("cannot load reviewed support helper")
    module = importlib.util.module_from_spec(specification)
    specification.loader.exec_module(module)
    return module

def free_bytes(path: Path) -> int:
    return shutil.disk_usage(path).free


def write_json(path: Path, value: object) -> None:
    path.write_text(json.dumps(value, indent=2, sort_keys=True) + "\n", encoding="utf-8")


def run_checked(command: list[str], *, env: dict[str, str], cwd: Path, timeout: int) -> subprocess.CompletedProcess[str]:
    return subprocess.run(command, cwd=cwd, env=env, text=True, capture_output=True, timeout=timeout, check=True)


def catalog_rows(performance: object, env: dict[str, str], port: int, database: str, query: str) -> list[str]:
    command = [str(performance.PSQL), "-X", "-A", "-t", "-v", "ON_ERROR_STOP=1", "-h", "127.0.0.1", "-p", str(port), "-U", ROLE, "-d", database, "-c", query]
    output = run_checked(command, env=env, cwd=SOURCE, timeout=20).stdout
    return [row for row in output.splitlines() if row]


def scrubbed_stream(value: str) -> dict[str, object]:
    encoded = value.encode("utf-8", errors="replace")
    return {"bytes": len(encoded), "sha256": hashlib.sha256(encoded).hexdigest(), "truncated": len(encoded) > MAX_STREAM_BYTES}


def make_policy(run_root: Path, port: int) -> Path:
    policy = run_root / "sheet-pg20.sb"
    content = "\n".join(
        (
            "(version 1)",
            "(allow default)",
            "(allow process*)",
            '(deny file-read* (subpath "/Users"))',
            "(deny network*)",
            f'(allow network-outbound (remote ip "localhost:{port}"))',
            "(deny file-write*)",
            f'(allow file-write* (subpath "{run_root}"))',
            '(allow file-write* (literal "/dev/null"))',
            '(deny file-write* (subpath ' + json.dumps(str(run_root / "guard")) + '))',
        )
    ) + "\n"
    policy.write_text(content, encoding="utf-8")
    return policy

def parse_callback(stdout: str) -> dict[str, object]:
    candidates = [line.removeprefix("SCHEMA_PG7_CALLBACK=") for line in stdout.splitlines() if line.startswith("SCHEMA_PG7_CALLBACK=")]
    if len(candidates) != 1:
        raise RuntimeError("test child did not emit exactly one safe callback")
    payload = json.loads(candidates[0])
    if not isinstance(payload, dict):
        raise RuntimeError("test child callback was malformed")
    return payload


def run_capped_child(command: list[str], *, cwd: Path, env: dict[str, str], timeout: int, performance: object) -> tuple[int, str, str, dict[str, object]]:
    process = subprocess.Popen(command, cwd=cwd, env=env, text=False, stdout=subprocess.PIPE, stderr=subprocess.PIPE, start_new_session=True)
    tracker = None
    captured: dict[str, bytearray] = {"stdout": bytearray(), "stderr": bytearray()}
    metadata: dict[str, dict[str, object]] = {"stdout": {"bytes": 0, "truncated": False}, "stderr": {"bytes": 0, "truncated": False}}

    def drain(name: str, stream: Any) -> None:
        digest = hashlib.sha256()
        while True:
            block = stream.read(65536)
            if not block:
                break
            digest.update(block)
            metadata[name]["bytes"] += len(block)
            if len(captured[name]) < MAX_STREAM_BYTES:
                captured[name].extend(block[:MAX_STREAM_BYTES - len(captured[name])])
            else:
                metadata[name]["truncated"] = True
        metadata[name]["sha256"] = digest.hexdigest()
        metadata[name]["truncated"] = metadata[name]["truncated"] or metadata[name]["bytes"] > MAX_STREAM_BYTES

    threads = [threading.Thread(target=drain, args=("stdout", process.stdout), daemon=True), threading.Thread(target=drain, args=("stderr", process.stderr), daemon=True)]
    started = time.monotonic()
    failure = None
    cleanup = None
    try:
        tracker = performance.OwnedProcesses(process)
        for thread in threads:
            thread.start()
        while True:
            tracker.observe()
            if process.poll() is not None:
                break
            if time.monotonic() - started > timeout:
                raise RuntimeError("bounded child deadline elapsed")
            if free_bytes(ROOT) < MIN_LIVE_BYTES:
                raise RuntimeError("child live disk floor")
            if any(item["bytes"] > MAX_STREAM_BYTES for item in metadata.values()):
                raise RuntimeError("child stream limit")
            time.sleep(0.05)
    except BaseException:
        failure = type(sys.exception()).__name__
        raise
    finally:
        if tracker is not None:
            cleanup = tracker.cleanup()
        elif process.poll() is None:
            process.terminate()
        process.wait(timeout=10)
        for thread in threads:
            if thread.ident is not None:
                thread.join(timeout=5)
        if failure is None and (cleanup is None or cleanup["status"] != "clean" or cleanup["signalled"] or any(thread.is_alive() for thread in threads)):
            raise RuntimeError("child process or stream cleanup incomplete")
    return process.returncode, captured["stdout"].decode("utf-8", errors="replace"), captured["stderr"].decode("utf-8", errors="replace"), {"process": cleanup, "streams": metadata}


def assert_test_result(result: dict[str, object], child_returncode: int) -> None:
    if child_returncode != 0 or result.get("status") != 0:
        raise RuntimeError("pytest did not exit cleanly")
    nodes = result.get("nodes")
    reports = result.get("reports")
    if nodes != list(NODES) or len(set(NODES)) != len(NODES):
        raise RuntimeError("collection did not contain exactly the frozen requested nodes")
    if result.get("collection_errors"):
        raise RuntimeError("collection errors occurred")
    if result.get("controls") != ["foreign_database", "foreign_port", "libpq_options", "inherited_libpq_env", "native_offport_os_denial", "guard_write_denied", "user_read_denied"]:
        raise RuntimeError("isolation controls incomplete")
    if not isinstance(reports, list) or len(reports) != len(NODES) * 3:
        raise RuntimeError("expected exactly three reports per frozen node")
    expected = {(node, phase) for node in NODES for phase in ("setup", "call", "teardown")}
    seen: set[tuple[str, str]] = set()
    for report in reports:
        if not isinstance(report, dict):
            raise RuntimeError("malformed report")
        key = (report.get("nodeid"), report.get("when"))
        if key not in expected or key in seen:
            raise RuntimeError("unexpected or duplicate phase report")
        if report.get("outcome") != "passed" or report.get("wasxfail"):
            raise RuntimeError("skip, xfail, or failure occurred")
        seen.add(key)
    if seen != expected:
        raise RuntimeError("phase coverage was incomplete")

def run_execute() -> dict[str, object]:

        "status": "started",

    environment = controlled_environment(performance, run_root, port, database)

    cluster_identity = None
    postmaster = None
    cluster_started = False
    completed = False

            cluster_identity, postmaster = performance.initialize_cluster(run_root, port, environment)
        except performance.ClusterInitializationError as error:
            candidate_process = getattr(error, "process", None)
            candidate_identity = getattr(error, "identity", None)
            if candidate_process is not None and isinstance(candidate_identity, dict) and candidate_identity.get("pid") == candidate_process.pid and candidate_identity.get("data_directory"):
                cluster_identity = candidate_identity
                postmaster = candidate_process
                cluster_started = True
                tracker = performance.OwnedProcesses(postmaster)
            raise
        cluster_started = True
        tracker = performance.OwnedProcesses(postmaster)
        result["postmaster"] = {"identity": cluster_identity, "birth": tracker.root}
        if free_bytes(Path("/private/tmp")) < MIN_LIVE_BYTES:
            raise RuntimeError("free disk fell below live-cluster minimum")
        run_checked([str(performance.PSQL), "-X", "-v", "ON_ERROR_STOP=1", "-h", "127.0.0.1", "-p", str(port), "-U", ROLE, "-d", "postgres", "-c", f"CREATE DATABASE {database}"], env=environment, cwd=SOURCE, timeout=20)
        pre_schemas = catalog_rows(performance, environment, port, database, "SELECT nspname FROM pg_namespace WHERE nspname ~ '" + OWNED_SCHEMA_PATTERN + "' ORDER BY 1")
        if pre_schemas:
            raise RuntimeError("fresh cluster unexpectedly contains owned test artifacts")
        guard_dir = run_root / "guard"
        guard_dir.mkdir(mode=0o700)
        shutil.copyfile(GUARD_SOURCE, guard_dir / "sitecustomize.py")
        write_json(guard_dir / "context.json", {"database": database, "host": "127.0.0.1", "port": port, "role": ROLE})
        policy = make_policy(run_root, port)
        child_env = dict(environment)
        child_env["PYTHONPATH"] = str(guard_dir)
        child_env["PYTEST_ADDOPTS"] = ""
        child_env["PYTEST_DISABLE_PLUGIN_AUTOLOAD"] = "1"
        command = ["sandbox-exec", "-f", str(policy), str(VENV_PYTHON), "-B", "-c", child_code()]
        child_timeout = max(1.0, MAX_SECONDS - (time.monotonic() - started) - 40)
        child_returncode, child_stdout, child_stderr, child_capture = run_capped_child(command, cwd=SOURCE, env=child_env, timeout=child_timeout, performance=performance)
        if time.monotonic() - started > MAX_SECONDS:
            raise RuntimeError("controller deadline exceeded")
        result["child"] = {"returncode": child_returncode, **child_capture}
        if child_capture["streams"]["stdout"]["truncated"] or child_capture["streams"]["stderr"]["truncated"]:
            raise RuntimeError("test output exceeded the active stream cap")
        callback = parse_callback(child_stdout)
        result["callback"] = callback
        assert_test_result(callback, child_returncode)
        result["post_test_schemas"] = catalog_rows(performance, environment, port, database, "SELECT nspname FROM pg_namespace WHERE nspname ~ '" + OWNED_SCHEMA_PATTERN + "' ORDER BY 1")
        if result["post_test_schemas"]:
            raise RuntimeError("tests left owned schemas")
        run_checked([str(performance.PSQL), "-X", "-v", "ON_ERROR_STOP=1", "-h", "127.0.0.1", "-p", str(port), "-U", ROLE, "-d", "postgres", "-c", f"DROP DATABASE {database}"], env=environment, cwd=SOURCE, timeout=20)
        result["generated_database_absent"] = catalog_rows(performance, environment, port, "postgres", f"SELECT datname FROM pg_database WHERE datname = '{database}'")
        if result["generated_database_absent"]:
            raise RuntimeError("generated database remains after drop")
        result["catalog_after_database_drop"] = catalog_rows(performance, environment, port, "postgres", "SELECT datname FROM pg_database ORDER BY 1")
        if result["catalog_after_database_drop"] != ["postgres", "template0", "template1"]:
            raise RuntimeError("post-drop database catalog is not system-only")
        result["source_after"] = source_manifest(SOURCE)
        if result["source_before"] != result["source_after"]:
            raise RuntimeError("immutable source changed during test run")
        result["controller_after_sha256"] = sha256_path(CONTROLLER)
        result["guard_after_sha256"] = sha256_path(GUARD_SOURCE)
        result["nodes_after_sha256"] = sha256_path(NODES_SOURCE)
        result["private_inputs_after"] = private_input_hashes()
        if result["controller_before_sha256"] != result["controller_after_sha256"] or result["guard_before_sha256"] != result["guard_after_sha256"] or result["nodes_before_sha256"] != result["nodes_after_sha256"]:
            raise RuntimeError("controller, guard, or frozen node list changed during test run")
        if result["private_inputs_before"] != result["private_inputs_after"]:
            raise RuntimeError("private overlay or private test source changed during test run")
        completed = True
    except BaseException:
        result["status"] = "failed"
        result["failure"] = {"type": type(sys.exception()).__name__}
        raise
    finally:
        try:
            if cluster_started and postmaster is not None and cluster_identity is not None:
                cleanup_observation: dict[str, object] = {}
                try:
                    cleanup_observation["database"] = catalog_rows(performance, environment, port, "postgres", f"SELECT datname FROM pg_database WHERE datname = '{database}'")
                    if cleanup_observation["database"]:
                        cleanup_observation["schemas"] = catalog_rows(performance, environment, port, database, "SELECT nspname FROM pg_namespace WHERE nspname ~ '" + OWNED_SCHEMA_PATTERN + "' ORDER BY 1")
                    else:
                        cleanup_observation["schema_database_already_absent"] = True
                except BaseException as error:
                    result["status"] = "failed"
                    cleanup_observation["catalog_error"] = type(error).__name__
                result["cleanup_observation"] = cleanup_observation
                if cleanup_observation.get("database"):
                    try:
                        run_checked([str(performance.PSQL), "-X", "-v", "ON_ERROR_STOP=1", "-h", "127.0.0.1", "-p", str(port), "-U", ROLE, "-d", "postgres", "-c", f"DROP DATABASE {database}"], env=environment, cwd=SOURCE, timeout=20)
                        result["cleanup_database_absent"] = catalog_rows(performance, environment, port, "postgres", f"SELECT datname FROM pg_database WHERE datname = '{database}'")
                    except BaseException:
                        result["status"] = "failed"
                        result["cleanup_database_error"] = type(sys.exception()).__name__
                if tracker is not None and tracker.root is not None and tracker.live(tracker.root):
                    result["stop"] = performance.stop_exact_cluster(
                        Path(cluster_identity["data_directory"]),
                        postmaster.pid,
                        postmaster,
                    )
                elif postmaster.poll() is not None:
                    result["stop"] = {"status": "already_exited"}
                elif tracker is None:
                    result["stop"] = performance.stop_exact_cluster(Path(cluster_identity["data_directory"]), postmaster.pid, postmaster)
                else:
                    raise RuntimeError("owned postmaster birth identity changed before stop")
                if result["stop"].get("status") not in {"stopped", "already_exited"}:
                    raise RuntimeError("exact postmaster stop was refused")
                stop_verified = True
        except BaseException:
            result["status"] = "failed"
            result["cleanup_error"] = type(sys.exception()).__name__
            raise
        finally:
            if tracker is not None:
                tracker.observe()
                result["owned_processes"] = {
                    "birth": tracker.root,
                    "remaining": tracker.remaining(),
                }
                if result["owned_processes"]["remaining"]:
                    result["status"] = "failed"
            if completed and stop_verified and result["status"] == "started":
                result["status"] = "passed"
            result["finished_at"] = time.time()
            write_json(run_root / "result.json", result)
    return result


def plan() -> dict[str, object]:
    return {
        "action": "preparation_only",
        "bridge_paths": list(BRIDGED_PATHS),
        "controller_sha256": sha256_path(CONTROLLER),
        "guard_sha256": sha256_path(GUARD_SOURCE),
        "nodes_sha256": sha256_path(NODES_SOURCE),
        "reviewed_helpers": {
            "native_lifecycle_sha256": sha256_path(PERFORMANCE_HELPER),
            "owned_processes_sha256": sha256_path(OWNED_HELPER),
        },
        "private_test_copies": {
            str(PRIVATE_RUNTIME_TEST): EXPECTED_RUNTIME_TEST_SHA256,
            str(PRIVATE_ADAPTER_TEST): EXPECTED_ADAPTER_TEST_SHA256,
        },
        "executor_overlay": {
            "path": str(EXECUTOR_OVERLAY),
            "base_sha256": EXPECTED_BASE_EXECUTOR_SHA256,
            "candidate_sha256": EXPECTED_CANDIDATE_EXECUTOR_SHA256,
            "contract": "frozen base with exactly two JSONB operator substitutions",
        },
        "immutable_source": str(SOURCE),
        "nodes": list(NODES),
        "required_free_bytes": {"before_start": MIN_START_BYTES, "live": MIN_LIVE_BYTES},
        "tmux": f"tmux new-session -s audit-sheet-wrapper-pg20-v1 '{VENV_PYTHON} -B {CONTROLLER} --execute'",
    }


def main() -> int:
    parser = argparse.ArgumentParser()
    parser.add_argument("--execute", action="store_true")
    arguments = parser.parse_args()
    if not arguments.execute:
        print(json.dumps(plan(), indent=2, sort_keys=True))
        return 0
    result = run_execute()
    print(json.dumps({"result": str(Path(result["run_root"]) / "result.json"), "status": result["status"]}, sort_keys=True))
    return 0 if result["status"] == "passed" else 1


if __name__ == "__main__":
    raise SystemExit(main())
