"""Guard for the disposable native-PostgreSQL Sheets test profile.

This module is copied as ``sitecustomize.py`` into a per-run directory.  It
accepts only the controller's one loopback PostgreSQL endpoint and database.
It is deliberately independent from application source code.
"""

import json
import os
import re
import sys
from pathlib import Path


ROLE = "readiness_perf_owner"
DATABASE_PATTERN = re.compile(r"readiness_sheet_test_[0-9a-f]{32}")
CONTEXT_KEYS = {"database", "host", "port", "role"}
DSN_KEYS = {
    "application_name",
    "connect_timeout",
    "dbname",
    "host",
    "port",
    "sslmode",
    "user",
}


def validate_context(candidate: object) -> dict[str, object]:
    if not isinstance(candidate, dict) or set(candidate) != CONTEXT_KEYS:
        raise PermissionError("invalid RBAC runtime context")
    if candidate["host"] != "127.0.0.1" or candidate["role"] != ROLE:
        raise PermissionError("unadmitted RBAC runtime identity")
    if not isinstance(candidate["port"], int) or not (20000 <= candidate["port"] <= 65535):
        raise PermissionError("unadmitted RBAC runtime port")
    database = candidate["database"]
    if not isinstance(database, str) or not DATABASE_PATTERN.fullmatch(database):
        raise PermissionError("unadmitted RBAC runtime database")
    return candidate


def validate_parameters(parameters: dict[str, str], context: dict[str, object]) -> None:
    if set(parameters) - DSN_KEYS:
        raise PermissionError("unexpected libpq parameter")
    if (
        parameters.get("host") != context["host"]
        or parameters.get("port") != str(context["port"])
        or parameters.get("user") != ROLE
        or parameters.get("dbname") != context["database"]
    ):
        raise PermissionError("connection is outside the disposable schema database")
    if parameters.get("sslmode", "disable") not in {"disable", "prefer"}:
        raise PermissionError("unexpected TLS configuration")


def validate_environment() -> None:
    if any(name.startswith("PG") for name in os.environ):
        raise PermissionError("inherited libpq environment is prohibited")


def install(context: dict[str, object]) -> None:
    import psycopg2
    from psycopg2.extensions import make_dsn, parse_dsn

    original_connect = psycopg2.connect

    def guarded_connect(dsn=None, connection_factory=None, cursor_factory=None, **keywords):
        validate_environment()
        parameters = parse_dsn(make_dsn(dsn, **keywords))
        validate_parameters(parameters, context)
        parameters["connect_timeout"] = "5"
        parameters["sslmode"] = "disable"
        parameters["options"] = "-c statement_timeout=30000 -c lock_timeout=5000"
        return original_connect(
            make_dsn(**parameters),
            connection_factory=connection_factory,
            cursor_factory=cursor_factory,
        )

    psycopg2.connect = guarded_connect


    def audit(event, arguments):
        if event == "socket.connect":
            address = arguments[1]
            if not isinstance(address, tuple) or address[:2] != (context["host"], context["port"]):
                raise PermissionError("RBAC fixture prohibits other network connections")
        elif event == "socket.getaddrinfo" and arguments[0] != context["host"]:
            raise PermissionError("RBAC fixture prohibits external name resolution")
        elif event in {"socket.gethostbyname", "socket.gethostbyaddr", "socket.sendto", "socket.bind"}:
            raise PermissionError("RBAC fixture prohibits this network operation")

    sys.addaudithook(audit)
    os.environ["PYTHON_DOTENV_DISABLED"] = "1"


if __name__ == "sitecustomize":
    try:
        context_path = Path(__file__).resolve().with_name("context.json")
        if context_path.is_symlink() or not context_path.is_file():
            raise PermissionError("RBAC runtime context must be a regular file")
        validate_environment()
        install(validate_context(json.loads(context_path.read_text(encoding="utf-8"))))
    except BaseException:
        os.write(2, b"RBAC runtime guard initialization failed\n")
        os._exit(78)
