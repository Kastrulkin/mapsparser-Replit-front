#!/usr/bin/env python3
"""Pure regression proof for the Sheets claim query compatibility seam.

It imports DatabaseManager's real cursor wrapper from an immutable clean source
tree but invokes two explicitly pinned executor copies.  No connection factory,
provider adapter, app server, or test runner is involved.
"""

from __future__ import annotations

import argparse
import hashlib
import importlib.util
import json
import os
import re
import sys
from pathlib import Path
from typing import Any


ROOT = Path(__file__).resolve().parent
SOURCE = Path("/private/tmp/localos-current-full-v10.BYwJr6/source")
CLEAN_EXECUTOR = ROOT / "executor-clean.py"
DIRTY_EXECUTOR = ROOT / "executor-dirty.py"
RESULT = ROOT / "result.json"
EXPECTED = {
    "clean_executor": "f310241af7d9febbf3eeae1974b4abce94ecd4b865be0a1a81e1bc1164c4b9ba",
    "database_manager": "44eea9318d8a940e1cda9e85e22828f064c1cc883a776ce5f08d3ce2ec006775",
    "dirty_executor": "fd190611c27d1663e38f125c3d5ac79699776b2da46e078cc82b98635bbb7763",
    "query_adapter": "44a6528921071090dce10423f93e5e627abe0cf451e8ae2b8314db4e242414c3",
}
IMMUTABLE_INPUTS = {
    "database_manager": SOURCE / "src/database_manager.py",
    "query_adapter": SOURCE / "src/query_adapter.py",
    "agent_api_security": SOURCE / "src/core/agent_api_security.py",
    "google_adapter": SOURCE / "src/services/agent_google_sheets_adapter.py",
}


def sha256_path(path: Path) -> str:
    digest = hashlib.sha256()
    with path.open("rb") as source:
        for block in iter(lambda: source.read(1024 * 1024), b""):
            digest.update(block)
    return digest.hexdigest()


def source_manifest() -> dict[str, str]:
    manifest: dict[str, str] = {}
    for name, path in IMMUTABLE_INPUTS.items():
        if not path.is_file() or path.is_symlink():
            raise RuntimeError("immutable proof input is unavailable")
        manifest[name] = sha256_path(path)
    return manifest


def source_tree_manifest() -> dict[str, object]:
    records: list[tuple[str, str]] = []
    for path in sorted(SOURCE.rglob("*")):
        if path.is_symlink():
            raise RuntimeError("immutable source tree contains a symlink")
        if path.is_file():
            records.append((str(path.relative_to(SOURCE)), sha256_path(path)))
    encoded = json.dumps(records, separators=(",", ":"), ensure_ascii=False).encode("utf-8")
    return {"count": len(records), "sha256": hashlib.sha256(encoded).hexdigest()}


def write_json(value: object) -> None:
    RESULT.write_text(json.dumps(value, indent=2, sort_keys=True) + "\n", encoding="utf-8")


def bind_inputs() -> dict[str, str]:
    values = {
        "clean_executor": sha256_path(CLEAN_EXECUTOR),
        "database_manager": sha256_path(IMMUTABLE_INPUTS["database_manager"]),
        "dirty_executor": sha256_path(DIRTY_EXECUTOR),
        "query_adapter": sha256_path(IMMUTABLE_INPUTS["query_adapter"]),
    }
    if values != EXPECTED:
        raise RuntimeError("proof inputs do not match their reviewed hashes")
    return values


class RecordingCursor:
    rowcount = 0

    def __init__(self) -> None:
        self.queries: list[tuple[str, object]] = []

    def execute(self, query: str, params: object = None) -> None:
        self.queries.append((query, params))

    def fetchone(self) -> None:
        return None


def load_executor(name: str, path: Path) -> object:
    specification = importlib.util.spec_from_file_location(name, path)
    if specification is None or specification.loader is None:
        raise RuntimeError("executor copy cannot be loaded")
    module = importlib.util.module_from_spec(specification)
    specification.loader.exec_module(module)
    return module


def verify_clean_red(wrapper: object, executor: object) -> dict[str, object]:
    raw = RecordingCursor()
    cursor = wrapper(raw)
    try:
        executor.claim_next_sheet_provider_request(cursor, business_id="")
    except ValueError:
        error = sys.exception()
        if "Found 1 bind placeholders, but 4 params provided" not in str(error):
            raise RuntimeError("clean failure reason differs from the adapter mismatch")
    else:
        raise RuntimeError("clean executor unexpectedly bypassed the query adapter")
    if len(raw.queries) != 2:
        raise RuntimeError("clean executor did not fail at its parameterized third execute")
    return {"raw_execute_count": len(raw.queries), "status": "expected_query_adapter_value_error"}


def verify_dirty_green(wrapper: object, executor: object) -> dict[str, object]:
    raw = RecordingCursor()
    cursor = wrapper(raw)
    if executor.claim_next_sheet_provider_request(cursor, business_id="") is not None:
        raise RuntimeError("recording cursor must not produce a claim")
    if len(raw.queries) != 3:
        raise RuntimeError("dirty executor did not reach exactly three raw executes")
    hold_query, hold_parameters = raw.queries[0]
    claim_query, claim_parameters = raw.queries[2]
    predicate = "jsonb_exists(approval.payload_json, 'sheet_write_snapshot')"
    if predicate not in hold_query or predicate not in claim_query:
        raise RuntimeError("JSONB key-existence approval predicate was not preserved")
    if "?" in claim_query:
        raise RuntimeError("parameterized claim query still contains an adapter-triggering question mark")
    if not isinstance(claim_parameters, tuple) or len(claim_parameters) != 4:
        raise RuntimeError("claim parameter tuple changed")
    business_one, business_two, lease_token, lease_seconds = claim_parameters
    if business_one != "" or business_two != "" or lease_seconds != 300 or not isinstance(lease_token, str) or re.fullmatch(r"[0-9a-f-]{36}", lease_token) is None:
        raise RuntimeError("claim parameter semantics changed")
    if hold_parameters is not None:
        raise RuntimeError("approval hold query unexpectedly gained parameters")
    return {"raw_execute_count": len(raw.queries), "status": "expected_three_executes"}


def execute() -> dict[str, object]:
    if os.environ.get("PYTHON_DOTENV_DISABLED") != "1":
        raise RuntimeError("dotenv must be disabled for the pure proof")
    if str(SOURCE / "src") not in sys.path:
        sys.path.insert(0, str(SOURCE / "src"))
    inputs = bind_inputs()
    immutable_before = source_manifest()
    source_tree_before = source_tree_manifest()
    from database_manager import DBCursorWrapper

    clean = load_executor("sheet_executor_clean_proof", CLEAN_EXECUTOR)
    dirty = load_executor("sheet_executor_dirty_proof", DIRTY_EXECUTOR)
    result = {
        "clean": verify_clean_red(DBCursorWrapper, clean),
        "dirty": verify_dirty_green(DBCursorWrapper, dirty),
        "inputs": inputs,
        "immutable_before": immutable_before,
    }
    result["immutable_after"] = source_manifest()
    result["source_tree_before"] = source_tree_before
    result["source_tree_after"] = source_tree_manifest()
    if result["immutable_before"] != result["immutable_after"] or result["source_tree_before"] != result["source_tree_after"]:
        raise RuntimeError("immutable source inputs changed during pure proof")
    result["status"] = "passed"
    return result


def main() -> int:
    parser = argparse.ArgumentParser()
    parser.add_argument("--execute", action="store_true")
    arguments = parser.parse_args()
    if not arguments.execute:
        print(json.dumps({"action": "preparation_only", "inputs": EXPECTED, "runner": str(Path(__file__).resolve())}, sort_keys=True))
        return 0
    result: dict[str, Any] = {"status": "failed"}
    try:
        result = execute()
        return 0
    except BaseException:
        result["failure"] = {"type": type(sys.exception()).__name__}
        raise
    finally:
        write_json(result)


if __name__ == "__main__":
    raise SystemExit(main())
