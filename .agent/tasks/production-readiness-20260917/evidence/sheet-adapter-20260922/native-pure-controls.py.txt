"""Pure Sheets profile accounting and admission checks; no app/DB/network."""

import copy
import importlib.util
from pathlib import Path


def load(name, filename):
    spec = importlib.util.spec_from_file_location(name, Path(__file__).with_name(filename))
    module = importlib.util.module_from_spec(spec)
    spec.loader.exec_module(module)
    return module


def rejected(call):
    try:
        call()
    except (RuntimeError, PermissionError):
        return
    raise AssertionError("invalid input was accepted")


runner = load("sheet_runner", "run_sheet_wrapper_pg20.py")
guard = load("sheet_guard", "sheet_pg20_sitecustomize.py")
context = {"database": "readiness_sheet_test_" + "0" * 32, "host": "127.0.0.1", "port": 64000, "role": guard.ROLE}
assert guard.validate_context(context) == context
parameters = {"dbname": context["database"], "host": "127.0.0.1", "port": "64000", "user": guard.ROLE}
guard.validate_parameters(parameters, context)
checks = 2
for key, value in (("host", "192.0.2.1"), ("port", "1"), ("dbname", "other"), ("user", "other"), ("options", "anything")):
    rejected(lambda: guard.validate_parameters({**parameters, key: value}, context))
    checks += 1
controls = ["foreign_database", "foreign_port", "libpq_options", "inherited_libpq_env", "native_offport_os_denial", "guard_write_denied", "user_read_denied"]
good = {"status": 0, "nodes": list(runner.NODES), "collection_errors": [], "controls": controls, "reports": [{"nodeid": node, "when": phase, "outcome": "passed", "wasxfail": False} for node in runner.NODES for phase in ("setup", "call", "teardown")]}
runner.assert_test_result(good, 0)
checks += 1
rejected(lambda: runner.assert_test_result(good, 1))
checks += 1
for key, value in (("outcome", "skipped"), ("wasxfail", True), ("nodeid", "different")):
    bad = copy.deepcopy(good)
    bad["reports"][0][key] = value
    rejected(lambda: runner.assert_test_result(bad, 0))
    checks += 1
for key, value in (("collection_errors", ["bad"]), ("reports", good["reports"][:-1]), ("controls", controls[:-1]), ("nodes", good["nodes"][:-1]), ("reports", [good["reports"][0]] * len(good["reports"]))):
    rejected(lambda: runner.assert_test_result({**good, key: value}, 0))
    checks += 1
assert len(runner.NODES) == 20
assert len(good["reports"]) == 60
checks += 2
print(f"Pure controls: {checks} passed; no PostgreSQL/app/network invoked")
