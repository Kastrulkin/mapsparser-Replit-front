"""Two unchanged browser regressions on a clean source snapshot, mocked APIs."""
import hashlib
import json
import os
from pathlib import Path
import signal
import subprocess
import time

ROOT = Path('/private/tmp/localos-browser-errors.gtS5JG')
SOURCE = ROOT / 'source'
NATIVE = Path('/private/tmp/localos-readiness-20260921.hfLYPi/native')
PYTHON = NATIVE / 'venv/bin/python'
TARGETS = [
    'tests/e2e/test_guided_tour_transient_gateway_error.py',
    'tests/e2e/test_telegram_mini_app_operator_error.py',
]

def manifest():
    rows = []
    for path in sorted(SOURCE.rglob('*')):
        if 'node_modules' in path.parts or not path.is_file():
            continue
        rows.append([str(path.relative_to(SOURCE)), hashlib.sha256(path.read_bytes()).hexdigest()])
    return rows

def main():
    destination = ROOT / 'evidence/browser-v1.json'
    if destination.exists():
        raise RuntimeError('attempt already exists')
    before = manifest()
    writable = ROOT / 'writable'
    environment = {
        'PATH': '/usr/local/opt/node@22/bin:/usr/local/bin:/usr/bin:/bin',
        'HOME': str(writable), 'TMPDIR': str(writable),
        'XDG_CACHE_HOME': str(writable),
        'PLAYWRIGHT_BROWSERS_PATH': str(NATIVE / 'ms-playwright'),
        'PYTHONDONTWRITEBYTECODE': '1', 'PYTHONNOUSERSITE': '1',
        'PYTHON_DOTENV_DISABLED': '1', 'PYTEST_DISABLE_PLUGIN_AUTOLOAD': '1',
        'NPM_CONFIG_CACHE': str(writable / 'npm-cache'),
        'NPM_CONFIG_UPDATE_NOTIFIER': 'false', 'NPM_CONFIG_AUDIT': 'false',
    }
    policy = '\n'.join([
        '(version 1)', '(allow default)', '(deny network*)',
        '(allow network-bind (local ip "127.0.0.1:*"))',
        '(allow network-inbound (local ip "127.0.0.1:*"))',
        '(allow network-outbound (remote ip "127.0.0.1:*"))',
        '(deny file-write*)',
        '(deny file-read-data (subpath "/Users") (subpath "/private/tmp"))',
        *['(allow file-read-data (subpath ' + json.dumps(str(path)) + '))'
          for path in (SOURCE, writable, NATIVE / 'venv', NATIVE / 'ms-playwright')],
        '(allow file-write* (subpath ' + json.dumps(str(writable)) + '))',
        '(allow file-write* (subpath ' + json.dumps(str(SOURCE / 'frontend/node_modules/.vite')) + '))',
        '(allow file-write* (subpath ' + json.dumps(str(SOURCE / 'frontend/node_modules/.vite-temp')) + '))',
        '(allow file-write* (literal "/dev/null"))',
    ])
    payload = '''
import json, socket, sys
sys.path.insert(0, %r)
import pytest
state = {'nodes': [], 'stages': []}
class Results:
    def pytest_collection_finish(self, session):
        state['nodes'] = [item.nodeid for item in session.items]
    def pytest_runtest_logreport(self, report):
        state['stages'].append({'node': report.nodeid, 'when': report.when, 'outcome': report.outcome, 'xfail': bool(getattr(report, 'wasxfail', None))})
result = pytest.main(%r + ['-q', '-p', 'no:cacheprovider'], plugins=[Results()])
state['exit'] = int(result)
print('BROWSER_RESULT=' + json.dumps(state, sort_keys=True))
raise SystemExit(result)
''' % (str(SOURCE), TARGETS)
    command = ['/usr/bin/sandbox-exec', '-p', policy, '/usr/bin/arch', '-arm64', str(PYTHON), '-I', '-B', '-c', payload]
    started = time.monotonic()
    process = subprocess.Popen(command, cwd=SOURCE, env=environment, text=True, stdout=subprocess.PIPE, stderr=subprocess.PIPE, start_new_session=True)
    timed_out = False
    try:
        stdout, stderr = process.communicate(timeout=180)
    except subprocess.TimeoutExpired:
        timed_out = True
        os.killpg(process.pid, signal.SIGTERM)
        try:
            stdout, stderr = process.communicate(timeout=12)
        except subprocess.TimeoutExpired:
            os.killpg(process.pid, signal.SIGKILL)
            stdout, stderr = process.communicate(timeout=12)
    after = manifest()
    record = {'command': command, 'env_keys': sorted(environment), 'source_commit': 'ef6d3e1e',
              'before': before, 'after': after, 'source_unchanged': before == after,
              'exit_code': process.returncode, 'timed_out': timed_out,
              'duration_seconds': round(time.monotonic() - started, 3), 'stdout': stdout, 'stderr': stderr}
    destination.write_text(json.dumps(record, ensure_ascii=False, indent=2))
    print(json.dumps({key: record[key] for key in ('exit_code', 'timed_out', 'source_unchanged', 'duration_seconds')}))
    return 0 if process.returncode == 0 and not timed_out and before == after else 1

if __name__ == '__main__':
    raise SystemExit(main())
