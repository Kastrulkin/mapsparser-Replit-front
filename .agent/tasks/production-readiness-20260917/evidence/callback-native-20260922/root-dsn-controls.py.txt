def definitions(path, names, namespace):
    payload = path.read_bytes()
    tree = ast.parse(payload)
    selected = [node for node in tree.body if isinstance(node, ast.FunctionDef) and node.name in names]
    assert {node.name for node in selected} == set(names)
    exec(compile(ast.Module(body=selected, type_ignores=[]), str(path), 'exec'), namespace)
    return hashlib.sha256(payload).hexdigest()

def main():

    calls = []
    events = []
    capabilities = []
    def intercepted(dsn=None, **kwargs):
        factory = kwargs.pop('connection_factory', None)
        cursor = kwargs.pop('cursor_factory', None)
        actual = psycopg2.extensions.parse_dsn(psycopg2.extensions.make_dsn(dsn, **kwargs))
        calls.append({'actual': actual, 'factory': factory, 'cursor': cursor})
        return 'intercepted-libpq'
