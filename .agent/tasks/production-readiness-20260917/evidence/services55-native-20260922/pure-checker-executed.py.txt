def load(name, path):
    spec = importlib.util.spec_from_file_location(name, path)
    module = importlib.util.module_from_spec(spec)
    spec.loader.exec_module(module)
    return module

def denied(label, operation, error_class):
    try:
        operation()

        checks.append(label)

class FakeChild:
    def __init__(self, status):
        self.returncode = status
        self.stdout = io.BytesIO(b"")
        self.stderr = io.BytesIO(b"")
        self.terminated = False

    def poll(self):
        return self.returncode

    def terminate(self):
        self.terminated = True
        self.returncode = -15

    def wait(self, timeout):
        assert self.returncode is not None
        return self.returncode


def failed_tracker(_process):
    raise ValueError("synthetic tracker construction failure")


class BrokenStream:
    def read(self, _size):
        raise OSError("synthetic read failure")


original_popen = runner.subprocess.Popen
try:
    fake = FakeChild(None)
    runner.subprocess.Popen = lambda *_args, **_kwargs: fake
    denied("tracker_failure_stops_direct_child", lambda: runner.run_capped_child([], cwd=ROOT, env={}, timeout=1, performance=SimpleNamespace(OwnedProcesses=failed_tracker)), ValueError)
    assert fake.terminated and fake.returncode == -15
    fake = FakeChild(0)
    fake.stderr = BrokenStream()
    tracker = SimpleNamespace(observe=lambda: None, cleanup=lambda: {"status": "clean", "signalled": []})

finally:
    runner.subprocess.Popen = original_popen

    def wait(self, timeout):
        if self.returncode is None:

        return self.returncode
