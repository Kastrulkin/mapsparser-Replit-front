def sha(path):
    return hashlib.sha256(path.read_bytes()).hexdigest()

def require(condition, label):
    if not condition:
        raise RuntimeError(label)

def manifest():
    require(not any(p.is_symlink() for p in SOURCE.rglob('*')), 'source_symlink')
    return [[str(p.relative_to(SOURCE)), sha(p), p.stat().st_size, p.stat().st_mode & 0o777]
            for p in sorted(SOURCE.rglob('*')) if p.is_file()]

def invoke(code, limit):
    argv = ['/usr/bin/sandbox-exec', '-f', str(POLICY), '/usr/bin/arch',
            '-arm64', str(PYTHON), '-B', '-I', '-c', code]
    env = {'PATH': '/usr/local/bin:/usr/bin:/bin', 'PYTEST_DISABLE_PLUGIN_AUTOLOAD': '1',
           'PYTHONDONTWRITEBYTECODE': '1', 'PYTHONNOUSERSITE': '1',
           'PYTHON_DOTENV_DISABLED': '1', 'TMPDIR': str(WRITABLE),
           'TMP': str(WRITABLE), 'TEMP': str(WRITABLE),
           'XDG_CACHE_HOME': str(WRITABLE / 'cache'),
           'DATABASE_URL': 'postgresql+psycopg2://metadata_only@127.0.0.1:1/localos_metadata_only'}
    started = time.monotonic()
    from owned_processes import OwnedProcesses
    child = subprocess.Popen(argv, cwd=SOURCE, env=env, stdout=subprocess.PIPE,
                             stderr=subprocess.PIPE, start_new_session=True)
    try:
        ownership = OwnedProcesses(child)
    except BaseException:
        child.kill()
        child.wait(timeout=5)
        raise
    buffers = {'stdout': bytearray(), 'stderr': bytearray()}
    counts = {'stdout': 0, 'stderr': 0}
    errors = []
    def drain(label, stream):
        try:
            while True:
                block = stream.read(8192)
                if not block:
                    break
                counts[label] += len(block)
                buffers[label].extend(block[:max(0, CAP - len(buffers[label]))])
        except BaseException:
            errors.append(label)
    threads = [threading.Thread(target=drain, args=(label, pipe), daemon=True)
               for label, pipe in [('stdout', child.stdout), ('stderr', child.stderr)]]
    for thread in threads:
        thread.start()
    reason = None
    try:
        while child.poll() is None:
            ownership.observe()
            if time.monotonic() - started > limit:
                reason = 'timeout'
            if shutil.disk_usage(ROOT).free < 2 * 1024**3:
                reason = 'disk_floor'
            if errors or any(value > CAP for value in counts.values()):
                reason = 'capture_failure'
            if reason:
                break
            time.sleep(.2)
    finally:
        cleanup = ownership.cleanup()
        child.wait(timeout=5)
        for thread in threads:
            thread.join(timeout=5)
    require(not errors and not any(thread.is_alive() for thread in threads), 'pipe_cleanup')
    require(cleanup['status'] == 'clean', 'owned_process_cleanup')
    metadata = {'exit_code': child.returncode, 'stopped_reason': reason,
                'process_cleanup': cleanup,
                'duration_seconds': round(time.monotonic() - started, 3),
                'argv': argv, 'streams': {label: {'total_bytes': counts[label],
                    'captured_bytes': len(value), 'sha256': hashlib.sha256(value).hexdigest(),
                    'truncated': counts[label] != len(value)} for label, value in buffers.items()}}
    return metadata, bytes(buffers['stdout']).decode('utf-8', errors='strict')

def account(expected, data, capture):
    reasons = []
    if capture['exit_code'] != 0 or capture['stopped_reason'] or data.get('return_code') != 0:
        reasons.append('process_nonpass')
    if data.get('nodes') != expected or len(set(expected)) != len(expected):
        reasons.append('node_drift')
    if data.get('collection_errors'):
        reasons.append('collection_error')
    stages = collections.Counter((row['nodeid'], row['when']) for row in data['reports'])
    desired = collections.Counter((node, when) for node in expected for when in ('setup', 'call', 'teardown'))
    if stages != desired:
        reasons.append('missing_duplicate_or_unknown_stage')
    if any(row['outcome'] != 'passed' or row['xfail'] for row in data['reports'] + data['subtests']):
        reasons.append('failed_skipped_or_xfail')
    if any(row['nodeid'] not in expected for row in data['subtests']):
        reasons.append('unknown_subtest')
    if any(stream['truncated'] for stream in capture['streams'].values()):
        reasons.append('truncated')
    return reasons

def pure_controls():
    expected = ['fixture::test_one']
    base = {'nodes': expected, 'collection_errors': [], 'return_code': 0, 'subtests': [],
            'reports': [{'nodeid': expected[0], 'when': when, 'outcome': 'passed', 'xfail': False}
                        for when in ('setup', 'call', 'teardown')]}
    capture = {'exit_code': 0, 'stopped_reason': None, 'streams': {}}
    require(not account(expected, base, capture), 'pure_positive')
    for mode in ('missing', 'duplicate', 'skip', 'xfail', 'node', 'collection', 'subtest', 'exit'):
        changed = json.loads(json.dumps(base)); meta = dict(capture)
        if mode == 'missing': changed['reports'].pop()
        if mode == 'duplicate': changed['reports'].append(changed['reports'][0])
        if mode == 'skip': changed['reports'][1]['outcome'] = 'skipped'
        if mode == 'xfail': changed['reports'][1]['xfail'] = True
        if mode == 'node': changed['nodes'] = ['other']
        if mode == 'collection': changed['collection_errors'] = ['fixture']
        if mode == 'subtest': changed['subtests'] = [{'nodeid': expected[0], 'when': 'call', 'outcome': 'failed', 'xfail': False}]
        if mode == 'exit': meta['exit_code'] = 1
        require(bool(account(expected, changed, meta)), 'pure_' + mode)
    return 9

def put(path, value):
    descriptor = os.open(path, os.O_CREAT | os.O_EXCL | os.O_WRONLY, 0o600)
    try:
        raw = (json.dumps(value, indent=2, sort_keys=True) + '\n').encode()
        while raw:
            raw = raw[os.write(descriptor, raw):]
    finally:
        os.close(descriptor)
