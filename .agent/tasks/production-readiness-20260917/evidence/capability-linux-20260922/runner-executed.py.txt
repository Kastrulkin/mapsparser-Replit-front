def manifest(root: Path) -> str:
    rows = []
    for path in sorted(candidate for candidate in root.rglob("*") if candidate.is_file()):
        rows.append(
            f"{path.relative_to(root)}\0{path.stat().st_mode & 0o777:o}\0{path.stat().st_size}\0"
            f"{hashlib.sha256(path.read_bytes()).hexdigest()}\n"
        )
    return hashlib.sha256("".join(rows).encode()).hexdigest()


sys.path[0:0] = [str(TESTDEPS), str(SOURCE), str(SOURCE / "src")]
import pytest


state = {"nodes": [], "reports": []}


class Collector:
    def pytest_collection_finish(self, session):
        state["nodes"] = [item.nodeid for item in session.items]

    def pytest_runtest_logreport(self, report):
        state["reports"].append(
            {
                "nodeid": report.nodeid,
                "when": report.when,
                "outcome": report.outcome,
                "xfail": bool(getattr(report, "wasxfail", False)),
            }
        )


source_before = manifest(SOURCE)
testdeps_before = manifest(TESTDEPS)
if source_before != EXPECTED_SOURCE or testdeps_before != EXPECTED_TESTDEPS:
    raise SystemExit("input_manifest_drift")

code = pytest.main(
    [
        *NODES,
        "-vv",
        "-p",
        "no:cacheprovider",
        "--tb=short",
        "--show-capture=no",
        "--basetemp",
        "/tmp/pytest",
    ],
    plugins=[Collector()],
)

expected_stages = collections.Counter(
    (nodeid, stage) for nodeid in NODES for stage in ("setup", "call", "teardown")
)
actual_stages = collections.Counter((row["nodeid"], row["when"]) for row in state["reports"])
reasons = []
if state["nodes"] != list(NODES):
    reasons.append("node_mismatch")
if actual_stages != expected_stages:
    reasons.append("stage_mismatch")
if code != 0:
    reasons.append("pytest_nonzero")
if any(row["outcome"] != "passed" or row["xfail"] for row in state["reports"]):
    reasons.append("nonpass_or_xfail")
source_after = manifest(SOURCE)
testdeps_after = manifest(TESTDEPS)
if source_after != source_before or testdeps_after != testdeps_before:
    reasons.append("input_drift")

print(
    "CAPABILITY_EXACT13="
    + json.dumps(
        {
            "node_count": len(state["nodes"]),
            "pytest_exit": int(code),
            "reasons": reasons,
            "reports": state["reports"],
            "source_manifest": source_after,
            "testdeps_manifest": testdeps_after,
        },
        sort_keys=True,
    )
)
raise SystemExit(0 if not reasons else 1)
