The helper reads only the six already-mapped dependency members from a pinned
local image save stream.  It does not extract layers, run containers, write
members, or retain source bytes, paths, markers, snippets, or credential
values.  A finding is never cleared merely because a parser rejects it: only
a separately reported positive structural predicate can support a later human
triage decision; everything else remains ``unknown``.
"""

BEGIN = re.compile(rb"-----BEGIN (?:[A-Z0-9]+ )?PRIVATE KEY-----")
BEGIN_LINE = re.compile(rb"^-----BEGIN ((?:[A-Z0-9]+ )?PRIVATE KEY)-----\r?$")
END_LINE = re.compile(rb"^-----END ((?:[A-Z0-9]+ )?PRIVATE KEY)-----\r?$")


def sha256_file(path):
    return hashlib.sha256(path.read_bytes()).hexdigest()


def dependency_hashes():
    return {
        "private_layer_helper": sha256_file(PRIVATE_PATH),
        "predicate_helper": sha256_file(Path(__file__)),
        "mapping": sha256_file(MAPPING),
    }


def line_number(data, offset):
    return data.count(b"\n", 0, offset) + 1


def line_bytes(data, wanted):
    require(wanted > 0, "invalid candidate line")
    start = 0
    current = 1
    while current < wanted:
        end = data.find(b"\n", start)
        if end < 0:
            raise ValueError("candidate line outside member")
        start = end + 1
        current += 1
    end = data.find(b"\n", start)
    return data[start:] if end < 0 else data[start:end]


def well_formed_private_pem_count(data):
    """Count only complete, base64-decodable PEM blocks without exposing bytes."""
    lines = data.splitlines()
    count = 0
    index = 0
    while index < len(lines):
        begin = BEGIN_LINE.fullmatch(lines[index])
        if begin is None:
            index += 1
            continue
        label = begin.group(1)
        body = []
        cursor = index + 1
        while cursor < len(lines):
            end = END_LINE.fullmatch(lines[cursor])
            if end is not None:
                if end.group(1) == label and body:
                    encoded = b"".join(body)
                    try:
                        decoded = base64.b64decode(encoded, validate=True)
                    except (ValueError, binascii.Error):
                        decoded = b""
                    if decoded:
                        count += 1
                index = cursor + 1
                break
            if not re.fullmatch(rb"[A-Za-z0-9+/]+={0,2}", lines[cursor]):
                index += 1
                break
            body.append(lines[cursor])
            cursor += 1
        else:
            index += 1
    return count


def source_literal_marker(data, candidate_lines):
    """True only for valid Python source whose candidate lines are literals."""
    try:
        text = data.decode("utf-8")
        tree = compile(text, "<in-memory-member>", "exec", flags=0, dont_inherit=True, optimize=0)
    except (SyntaxError, UnicodeDecodeError, ValueError):
        return False
    # Compile validates Python source but deliberately does not execute it.
    del tree
    import ast

    parsed = ast.parse(text, mode="exec")
    literal_lines = set()
    for node in ast.walk(parsed):
        if isinstance(node, ast.Constant) and isinstance(node.value, (str, bytes)):
            value = node.value.encode() if isinstance(node.value, str) else node.value
            if BEGIN.search(value) is not None:
                literal_lines.update(range(node.lineno, getattr(node, "end_lineno", node.lineno) + 1))
    return all(line in literal_lines for line in candidate_lines)


def code_constant_marker(value, seen):
    if id(value) in seen:
        return False
    seen.add(id(value))
    if isinstance(value, (str, bytes)):
        candidate = value.encode() if isinstance(value, str) else value
        return BEGIN.search(candidate) is not None
    if isinstance(value, type((lambda: None).__code__)):
        return any(code_constant_marker(item, seen) for item in value.co_consts)
    if isinstance(value, tuple):
        return any(code_constant_marker(item, seen) for item in value)
    return False


def pyc_constant_marker(data):
    if len(data) < 16:
        return False
    try:
        value = marshal.loads(data[16:])
    except (EOFError, TypeError, ValueError):
        return False
    return code_constant_marker(value, set())


def inspect_member(data, candidate_lines):
    require(len(data) <= MAX_MEMBER, "selected member exceeds in-memory limit")
    require(all(line <= data.count(b"\n") + 1 for line in candidate_lines), "candidate line outside member")
    match_lines = {line_number(data, match.start()) for match in BEGIN.finditer(data)}
    exact_line_marker = {line: BEGIN.search(line_bytes(data, line)) is not None for line in candidate_lines}
    is_elf = data.startswith(b"\x7fELF")
    is_pyc = data[:4] == importlib.util.MAGIC_NUMBER
    return {
        "whole_file_sha256": hashlib.sha256(data).hexdigest(),
        "byte_count": len(data),
        "candidate_line_marker": exact_line_marker,
        "candidate_line_in_marker_set": {line: line in match_lines for line in candidate_lines},
        "well_formed_private_pem_count": well_formed_private_pem_count(data),
        "is_elf": is_elf,
        "all_marker_matches_nul_delimited": bool(match_lines) and all(
            match.start() > 0 and match.end() < len(data) and data[match.start() - 1:match.start()] == b"\0" and data[match.end():match.end() + 1] == b"\0"
            for match in BEGIN.finditer(data)
        ),
        "is_pyc": is_pyc,
        "pyc_marshal_contains_marker_constant": pyc_constant_marker(data) if is_pyc else False,
        "python_source_literal_contains_marker": source_literal_marker(data, candidate_lines) if not is_elf and not is_pyc else False,
    }


def mapped_private_candidates(mapping):
    require(sha256_file(PRIVATE_PATH) == PRIVATE_SHA256, "private-layer helper hash mismatch")
    require(sha256_file(MAPPING) == MAPPING_SHA256, "mapping hash mismatch")
    require(mapping.get("status") == "mapped" and mapping.get("archive_sha256") == EXPECTED_ARCHIVE_SHA256, "unexpected mapping evidence")
    require(mapping.get("image") == IMAGE == EXPECTED_IMAGE, "image identity mismatch")
    rows = [row for row in mapping["mapped_candidates"] if row["kind"] == "layer" and row["rule_id"] == "private-key"]
    require(len(rows) == 17 and len({row["candidate_id"] for row in rows}) == 17, "unexpected private candidate count")
    result = {}
    for row in rows:
        key = (row["blob"], row["diff_id"], row["path_sha256"])
        require(re.fullmatch(r"sha256:[0-9a-f]{64}", row["blob"]) is not None, "invalid blob")
        require(re.fullmatch(r"sha256:[0-9a-f]{64}", row["diff_id"]) is not None, "invalid diff id")
        require(re.fullmatch(r"[0-9a-f]{64}", row["path_sha256"]) is not None, "invalid path hash")
        require(isinstance(row["line"], int) and row["line"] > 0, "invalid candidate line")
        require(isinstance(row["whole_file_sha256"], str) and re.fullmatch(r"[0-9a-f]{64}", row["whole_file_sha256"]) is not None, "invalid file hash")
        require(isinstance(row["byte_count"], int) and 0 <= row["byte_count"] <= MAX_MEMBER, "unexpected member size")
        result.setdefault(key, []).append(row)
    require(len(result) == 6, "unexpected selected member count")
    return result


def inspect_image():
    result = subprocess.run(DOCKER + ["image", "inspect", IMAGE], env=ENV, capture_output=True, timeout=30, check=True)
    require(len(result.stdout) <= MAX_JSON, "inspect output exceeds limit")
    rows = json.loads(result.stdout)
    require(len(rows) == 1 and rows[0]["Id"] == IMAGE and rows[0]["Os"] == "linux" and rows[0]["Architecture"] == "arm64", "image identity mismatch")
    ids = rows[0]["RootFS"]["Layers"]
    require(len(ids) == 20 and len(set(ids)) == 20, "unexpected inspected layer list")
    return ids


def selected_layer(source, expected_diff_id, targets):
    layer = Meter(source, MAX_LAYER)
    archive = tarfile.open(fileobj=layer, mode="r|")
    found = {}
    try:
        for entry in archive:
            name = normalized(entry.name)
            if not entry.isreg():
                continue
            path_sha256 = hashlib.sha256(name.encode()).hexdigest()
            stream = archive.extractfile(entry)
            require(stream is not None, "regular member content unavailable")
            if path_sha256 not in targets:
                drain(Meter(stream, entry.size))
                stream.close()
                continue
            require(path_sha256 not in found, "ambiguous selected layer member")
            require(0 <= entry.size <= MAX_MEMBER, "selected member exceeds in-memory limit")
            member = Meter(stream, entry.size)
            parts = []
            while True:
                part = member.read(CHUNK)
                if not part:
                    break
                parts.append(part)
            stream.close()
            require(member.count == entry.size, "selected member truncated")
            found[path_sha256] = b"".join(parts)
    finally:
        archive.close()
    drain(layer)
    require("sha256:" + layer.digest.hexdigest() == expected_diff_id, "selected layer diff id mismatch")
    require(set(found) == set(targets), "selected member missing")
    return found, layer.count


def scan_archive(source, expected_diff_ids, targets):
    measured = Meter(source, MAX_ARCHIVE)
    archive = tarfile.open(fileobj=measured, mode="r|")
    seen, metadata, layers, sizes, findings = set(), {}, {}, {}, {}
    total = 0
    by_blob = {}
    for (blob, diff_id, path_sha256), rows in targets.items():
        current = by_blob.setdefault(blob, (diff_id, {}))
        require(current[0] == diff_id, "same blob has conflicting diff ids")
        current[1][path_sha256] = rows
    try:
        for entry in archive:
            name = normalized(entry.name)
            require(name not in seen and len(seen) < 100, "duplicate or excessive OCI member")
            seen.add(name)
            if entry.isdir():
                require(name in {"blobs", "blobs/sha256"}, "unexpected OCI directory")
                continue
            require(entry.isreg() and entry.sparse is None, "nonregular OCI member")
            sizes[name] = entry.size
            match = BLOB.fullmatch(name)
            require(match is not None or name in {"index.json", "manifest.json", "oci-layout"}, "unexpected OCI member")
            content = Meter(archive.extractfile(entry), min(MAX_ARCHIVE, entry.size))
            prefix = content.read(2)
            if prefix == b"\x1f\x8b":
                require(match is not None, "nonblob gzip member")
                blob = "sha256:" + match.group(1)
                compressed = gzip.GzipFile(fileobj=Prefix(prefix, content), mode="rb")
                try:
                    if blob in by_blob:
                        diff_id, selected = by_blob[blob]
                        members, expanded = selected_layer(compressed, diff_id, selected)
                        for path_sha256, data in members.items():
                            findings[(blob, diff_id, path_sha256)] = (data, expanded)
                        row = {"diff_id": diff_id, "uncompressed_bytes": expanded}
                    else:
                        row = PRIVATE["scan_layer"](compressed)
                finally:
                    compressed.close()
                drain(content)
                total += row["uncompressed_bytes"]
                require(total <= MAX_TOTAL, "total expanded image exceeds limit")
                layers[name] = {"compressed_bytes": content.count, **row}
            else:
                require(entry.size <= MAX_JSON, "unknown large non-gzip blob")
                metadata[name] = json.loads(prefix + small_bytes(content))
            require(content.count == entry.size, "OCI member truncated")
            if match is not None:
                require(content.digest.hexdigest() == match.group(1), "OCI content digest mismatch")
    finally:
        archive.close()
    drain(measured)
    ordered = validate_graph(metadata, layers, expected_diff_ids, sizes)
    require(set(findings) == set(targets), "selected target missing")
    return {"archive_bytes": measured.count, "archive_sha256": measured.digest.hexdigest(), "expanded_bytes": total, "layers": ordered, "members": findings}


def safe_result(archive, targets):
    rows = []
    for key, candidates in sorted(targets.items()):
        data, expanded = archive["members"][key]
        lines = sorted({row["line"] for row in candidates})
        result = inspect_member(data, lines)
        for candidate in candidates:
            require(result["whole_file_sha256"] == candidate["whole_file_sha256"], "mapped whole-file hash mismatch")
            require(result["byte_count"] == candidate["byte_count"], "mapped byte count mismatch")
        rows.append({
            "blob": key[0],
            "diff_id": key[1],
            "path_sha256": key[2],
            "candidate_ids": sorted(row["candidate_id"] for row in candidates),
            "candidate_lines": lines,
            "whole_file_sha256": result["whole_file_sha256"],
            "byte_count": result["byte_count"],
            "expanded_layer_byte_count": expanded,
            "candidate_line_marker": {str(line): result["candidate_line_marker"][line] for line in lines},
            "candidate_line_in_marker_set": {str(line): result["candidate_line_in_marker_set"][line] for line in lines},
            "well_formed_private_pem_count": result["well_formed_private_pem_count"],
            "is_elf": result["is_elf"],
            "all_marker_matches_nul_delimited": result["all_marker_matches_nul_delimited"],
            "is_pyc": result["is_pyc"],
            "pyc_marshal_contains_marker_constant": result["pyc_marshal_contains_marker_constant"],
            "python_source_literal_contains_marker": result["python_source_literal_contains_marker"],
            "classification": "structural_observation_only",
        })
    require(sum(len(row["candidate_ids"]) for row in rows) == 17, "candidate accounting mismatch")
    return rows


def write_exclusive(path, value):
    flags = os.O_WRONLY | os.O_CREAT | os.O_EXCL
    descriptor = os.open(path, flags, 0o600)
    try:
        with os.fdopen(descriptor, "w", encoding="utf-8") as stream:
            json.dump(value, stream, sort_keys=True, indent=2)
            stream.write("\n")
    except BaseException:
        try:
            os.unlink(path)
        except FileNotFoundError:
            pass
        raise


def main():
    parser = argparse.ArgumentParser(description=__doc__)
    parser.add_argument("--attempt", required=True)
    args = parser.parse_args()
    require(re.fullmatch(r"v[1-9][0-9]*", args.attempt) is not None, "invalid attempt")
    require(EVIDENCE.is_dir() and not EVIDENCE.is_symlink() and EVIDENCE.resolve(strict=True) == EVIDENCE, "noncanonical evidence directory")
    require(shutil.disk_usage(EVIDENCE).free >= 5 * 1024**3, "image stream requires 5 GiB start floor")
    destination = EVIDENCE / ("image-private-key-predicates-" + args.attempt + ".json")
    require(not destination.exists() and not destination.is_symlink(), "evidence already exists")
    started = time.monotonic()
    result = {"attempt": args.attempt, "image": IMAGE, "scope": "17-existing-private-key-detector-candidates-only", "classification_limit": "structural_observation_only", "dependencies_before": dependency_hashes()}
    process = None
    timer = None
    expired = threading.Event()
    try:
        mapping = json.loads(MAPPING.read_text(encoding="utf-8"))
        targets = mapped_private_candidates(mapping)
        diff_ids = inspect_image()
        process = subprocess.Popen(DOCKER + ["image", "save", "--platform", "linux/arm64", IMAGE], env=ENV, stdout=subprocess.PIPE, stderr=subprocess.DEVNULL, start_new_session=True)

        def timeout():
            expired.set()
            if process.poll() is None:
                os.killpg(process.pid, signal.SIGKILL)

        timer = threading.Timer(EXPORT_TIMEOUT, timeout)
        timer.start()
        archive = scan_archive(process.stdout, diff_ids, targets)
        require(process.wait(timeout=10) == 0 and not expired.is_set(), "image export failed or timed out")
        require(archive["archive_sha256"] == EXPECTED_ARCHIVE_SHA256, "archive hash mismatch")
        require(inspect_image() == diff_ids, "image identity changed during scan")
        result.update({key: archive[key] for key in ("archive_bytes", "archive_sha256", "expanded_bytes")})
        result["members"] = safe_result(archive, targets)
        result["status"] = "completed_structural_observations"
    except BaseException:
        result["status"] = "failed"
        result["error_type"] = type(sys.exception()).__name__
    finally:
        if timer is not None:
            timer.cancel()
        if process is not None and process.poll() is None:
            os.killpg(process.pid, signal.SIGKILL)
            process.wait(timeout=10)
        result["dependencies_after"] = dependency_hashes()
        result["duration_seconds"] = round(time.monotonic() - started, 3)
        write_exclusive(destination, result)
    print(json.dumps({"status": result["status"], "output_sha256": sha256_file(destination), "candidate_count": 17}, sort_keys=True))


if __name__ == "__main__":
    main()
