"""Bounded structural observations for 17 mapped private-key detector candidates.

This v2 helper deliberately does not deserialize Python bytecode or parse any
untrusted executable format. It streams a pinned local image exactly once,
never extracts or runs it, and emits only hashes, booleans and counts.
Structural observations are not a secret-clearance decision.
"""

import argparse
import ast
import base64
import binascii
import hashlib
import importlib.util
import json
import os
from pathlib import Path
import re
import runpy
import shutil
import signal
import subprocess
import sys
import threading
import time


V1 = Path("/private/tmp/localos-image-pem-predicates-rcmSFg/image_private_key_predicates.py")
V1_SHA256 = "fe28298868e62ce1dc555a8a40ecbc90b1103a43e6eda24ccc34c082aea8d48c"
V1_MODULE = runpy.run_path(str(V1))
PRIVATE_PATH = V1_MODULE["PRIVATE_PATH"]
MAPPING = V1_MODULE["MAPPING"]
EVIDENCE = V1_MODULE["EVIDENCE"]
DOCKER = V1_MODULE["DOCKER"]
ENV = V1_MODULE["ENV"]
IMAGE = V1_MODULE["IMAGE"]
MAX_JSON = V1_MODULE["MAX_JSON"]
MAX_MEMBER = V1_MODULE["MAX_MEMBER"]
EXPORT_TIMEOUT = V1_MODULE["EXPORT_TIMEOUT"]
MAPPED_PRIVATE = V1_MODULE["mapped_private_candidates"]
INSPECT_IMAGE = V1_MODULE["inspect_image"]
SCAN_ARCHIVE = V1_MODULE["scan_archive"]

BEGIN = re.compile(rb"-----BEGIN (?:[A-Z0-9]+ )?PRIVATE KEY-----")
BEGIN_LINE = re.compile(rb"^-----BEGIN ((?:[A-Z0-9]+ )?PRIVATE KEY)-----\r?$")
END_LINE = re.compile(rb"^-----END ((?:[A-Z0-9]+ )?PRIVATE KEY)-----\r?$")
REGEX_META = b"[](){}?*+|\\"


def require(condition, message):
    if not condition:
        raise ValueError(message)


def digest(path):
    return hashlib.sha256(path.read_bytes()).hexdigest()


def dependencies():
    return {
        "v1_stream_helper": digest(V1),
        "private_layer_helper": digest(PRIVATE_PATH),
        "mapping": digest(MAPPING),
        "v2_predicate_helper": digest(Path(__file__)),
    }


def line_bytes(data, wanted):
    require(wanted > 0, "invalid candidate line")
    start = 0
    current = 1
    while current < wanted:
        end = data.find(b"\n", start)
        if end < 0:
            raise ValueError("candidate line outside member")
        start = end + 1
        current += 1
    end = data.find(b"\n", start)
    return data[start:] if end < 0 else data[start:end]


def line_number(data, offset):
    return data.count(b"\n", 0, offset) + 1


def complete_private_pem_blocks(data):
    """Recognize only complete base64-decodable PEM envelopes; retain no body."""
    count = 0
    index = 0
    lines = data.splitlines()
    while index < len(lines):
        begin = BEGIN_LINE.fullmatch(lines[index])
        if begin is None:
            index += 1
            continue
        label = begin.group(1)
        body = []
        cursor = index + 1
        while cursor < len(lines):
            end = END_LINE.fullmatch(lines[cursor])
            if end is not None:
                if end.group(1) == label and body:
                    try:
                        decoded = base64.b64decode(b"".join(body), validate=True)
                    except (ValueError, binascii.Error):
                        decoded = b""
                    if decoded:
                        count += 1
                index = cursor + 1
                break
            if re.fullmatch(rb"[A-Za-z0-9+/]+={0,2}", lines[cursor]) is None:
                index += 1
                break
            body.append(lines[cursor])
            cursor += 1
        else:
            index += 1
    return count


def literal_predicates(data, candidate_lines, complete_blocks):
    """Return positive, narrow source-literal observations; no bytecode decoding."""
    try:
        text = data.decode("utf-8")
        tree = ast.parse(text, mode="exec")
    except (SyntaxError, UnicodeDecodeError, ValueError):
        return {"python_exact_delimiter_literal": False, "python_bounded_regex_template_literal": False}
    exact_lines = set()
    template_lines = set()
    for node in ast.walk(tree):
        if not isinstance(node, ast.Constant) or not isinstance(node.value, (str, bytes)):
            continue
        value = node.value.encode() if isinstance(node.value, str) else node.value
        if len(value) > 4096 or BEGIN.search(value) is None:
            continue
        covered = range(node.lineno, getattr(node, "end_lineno", node.lineno) + 1)
        if BEGIN.fullmatch(value) is not None:
            exact_lines.update(covered)
        if complete_blocks == 0 and any(character in value for character in REGEX_META):
            try:
                re.compile(value)
            except re.error:
                continue
            template_lines.update(covered)
    return {
        "python_exact_delimiter_literal": all(line in exact_lines for line in candidate_lines),
        "python_bounded_regex_template_literal": all(line in template_lines for line in candidate_lines),
    }


def bounded_elf_marker(data, match):
    """A string-table delimiter requires NUL before and NUL after optional newline."""
    if match.start() == 0 or data[match.start() - 1:match.start()] != b"\0":
        return False
    cursor = match.end()
    if data[cursor:cursor + 2] == b"\r\n":
        cursor += 2
    elif data[cursor:cursor + 1] == b"\n":
        cursor += 1
    return cursor < len(data) and data[cursor:cursor + 1] == b"\0"


def inspect_member(data, candidate_lines):
    require(len(data) <= MAX_MEMBER, "selected member exceeds in-memory limit")
    require(all(line <= data.count(b"\n") + 1 for line in candidate_lines), "candidate line outside member")
    matches = list(BEGIN.finditer(data))
    line_marker = {line: BEGIN.search(line_bytes(data, line)) is not None for line in candidate_lines}
    complete_blocks = complete_private_pem_blocks(data)
    is_elf = data.startswith(b"\x7fELF")
    is_pyc = data[:4] == importlib.util.MAGIC_NUMBER
    line_bounded = {}
    for line in candidate_lines:
        on_line = [match for match in matches if line_number(data, match.start()) == line]
        line_bounded[line] = bool(on_line) and all(bounded_elf_marker(data, match) for match in on_line)
    return {
        "whole_file_sha256": hashlib.sha256(data).hexdigest(),
        "byte_count": len(data),
        "candidate_line_marker": line_marker,
        "complete_private_pem_block_count": complete_blocks,
        "is_elf": is_elf,
        "elf_candidate_markers_bounded_by_nul_delimiters": line_bounded if is_elf else {line: False for line in candidate_lines},
        "is_pyc_magic": is_pyc,
        "pyc_marker_observation_only": is_pyc and any(line_marker.values()),
        **literal_predicates(data, candidate_lines, complete_blocks),
    }


def result_rows(archive, targets):
    rows = []
    for key, candidates in sorted(targets.items()):
        data, expanded = archive["members"][key]
        lines = sorted({row["line"] for row in candidates})
        observed = inspect_member(data, lines)
        for candidate in candidates:
            require(observed["whole_file_sha256"] == candidate["whole_file_sha256"], "mapped whole-file hash mismatch")
            require(observed["byte_count"] == candidate["byte_count"], "mapped byte count mismatch")
        rows.append({
            "blob": key[0],
            "diff_id": key[1],
            "path_sha256": key[2],
            "candidate_ids": sorted(row["candidate_id"] for row in candidates),
            "candidate_lines": lines,
            "whole_file_sha256": observed["whole_file_sha256"],
            "byte_count": observed["byte_count"],
            "expanded_layer_byte_count": expanded,
            "candidate_line_marker": {str(line): observed["candidate_line_marker"][line] for line in lines},
            "complete_private_pem_block_count": observed["complete_private_pem_block_count"],
            "is_elf": observed["is_elf"],
            "elf_candidate_markers_bounded_by_nul_delimiters": {str(line): observed["elf_candidate_markers_bounded_by_nul_delimiters"][line] for line in lines},
            "is_pyc_magic": observed["is_pyc_magic"],
            "pyc_marker_observation_only": observed["pyc_marker_observation_only"],
            "python_exact_delimiter_literal": observed["python_exact_delimiter_literal"],
            "python_bounded_regex_template_literal": observed["python_bounded_regex_template_literal"],
            "classification": "structural_observation_only_unknown_not_cleared",
        })
    require(sum(len(row["candidate_ids"]) for row in rows) == 17, "candidate accounting mismatch")
    return rows


def write_exclusive(path, value):
    descriptor = os.open(path, os.O_WRONLY | os.O_CREAT | os.O_EXCL, 0o600)
    stream = os.fdopen(descriptor, "w", encoding="utf-8")
    try:
        json.dump(value, stream, sort_keys=True, indent=2)
        stream.write("\n")
        stream.flush()
        os.fsync(stream.fileno())
    finally:
        stream.close()


def main():
    parser = argparse.ArgumentParser(description=__doc__)
    parser.add_argument("--attempt", required=True)
    args = parser.parse_args()
    require(re.fullmatch(r"v[1-9][0-9]*", args.attempt) is not None, "invalid attempt")
    require(EVIDENCE.is_dir() and not EVIDENCE.is_symlink() and EVIDENCE.resolve(strict=True) == EVIDENCE, "noncanonical evidence directory")
    require(shutil.disk_usage(EVIDENCE).free >= 5 * 1024**3, "image stream requires 5 GiB start floor")
    destination = EVIDENCE / ("image-private-key-predicates-" + args.attempt + ".json")
    require(not destination.exists() and not destination.is_symlink(), "evidence already exists")
    started = time.monotonic()
    result = {
        "attempt": args.attempt,
        "image": IMAGE,
        "scope": "17-existing-private-key-detector-candidates-only",
        "classification_limit": "structural_observation_only_unknown_not_cleared",
        "dependencies_before": dependencies(),
    }
    process = None
    timer = None
    expired = threading.Event()
    try:
        require(digest(V1) == V1_SHA256, "v1 stream helper hash mismatch")
        mapping = json.loads(MAPPING.read_text(encoding="utf-8"))
        targets = MAPPED_PRIVATE(mapping)
        diff_ids = INSPECT_IMAGE()
        process = subprocess.Popen(DOCKER + ["image", "save", "--platform", "linux/arm64", IMAGE], env=ENV, stdout=subprocess.PIPE, stderr=subprocess.DEVNULL, start_new_session=True)

        def timeout():
            expired.set()
            if process.poll() is None:
                os.killpg(process.pid, signal.SIGKILL)

        timer = threading.Timer(EXPORT_TIMEOUT, timeout)
        timer.start()
        archive = SCAN_ARCHIVE(process.stdout, diff_ids, targets)
        require(process.wait(timeout=10) == 0 and not expired.is_set(), "image export failed or timed out")
        require(archive["archive_sha256"] == V1_MODULE["EXPECTED_ARCHIVE_SHA256"], "archive hash mismatch")
        require(INSPECT_IMAGE() == diff_ids, "image identity changed during scan")
        result.update({key: archive[key] for key in ("archive_bytes", "archive_sha256", "expanded_bytes")})
        result["members"] = result_rows(archive, targets)
        result["status"] = "completed_structural_observations"
    except BaseException:
        result["status"] = "failed"
        result["error_type"] = type(sys.exception()).__name__
    finally:
        if timer is not None:
            timer.cancel()
        if process is not None and process.poll() is None:
            os.killpg(process.pid, signal.SIGKILL)
            process.wait(timeout=10)
        result["dependencies_after"] = dependencies()
        if result["dependencies_after"] != result["dependencies_before"]:
            result["status"] = "failed"
            result["error_type"] = "DependencyDrift"
        result["duration_seconds"] = round(time.monotonic() - started, 3)
        write_exclusive(destination, result)
    print(json.dumps({"status": result["status"], "output_sha256": digest(destination), "candidate_count": 17}, sort_keys=True))
    if result["status"] == "failed":
        raise SystemExit(1)


if __name__ == "__main__":
    main()
