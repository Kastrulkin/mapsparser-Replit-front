"""Transfer prior source triage only through exact whole-file byte equality."""
import hashlib
import json
from pathlib import Path
import subprocess
import sys

ROOT = Path('/Users/alexdemyanov/Yandex.Disk-demyanovap.localized/Всякое/SEO с Реплит на Курсоре')
LEDGER = ROOT / '.agent/tasks/production-readiness-20260917/evidence/current-source-secret-triage-20260921.json'
assert hashlib.sha256(LEDGER.read_bytes()).hexdigest() == '5659a93122fac1a7b6e90f4f880558e73c0df8a02baec47d521b7825d213f92d'
ledger = json.loads(LEDGER.read_text())
assert ledger['commit'] == '334c9d4bf69cc9e96fd13a39d9eba8ff32c9f3d6'
assert ledger['confirmed_credentials'] == ledger['unresolved'] == 0
mapping_path = Path(sys.argv[1])
mapping = json.loads(mapping_path.read_text())
assert mapping['status'] == 'mapped'
assert len(mapping['mapped_candidates']) == 219
assert mapping['input_capture_sha256'] == '02fa7241514f55ae81ce184f6df88733696eca17927169b771dc0f4c4229a95e'
assert mapping['image'] == 'sha256:9d6edac8b6948e239c0bab54564f92f829e5b9dfc93f3ca0626060ffb4e60853'
assert mapping['dependency_hashes_before'] == mapping['dependency_hashes_after']
assert mapping['dependency_hashes_before'] == {
    'mapper': '3f55167f80900fc08e32484e5fe9b169ee50b3b2f29f12ab0df4b110d11f4a59',
    'private_layer_scanner': '57a1ba58876f35797708071ba88b2387a0e15888ac0f7034868cbb017da3a8f7',
}
assert mapping['archive_sha256'] == '35d4ad6e70b812698c42698c6c93f16256da4ef3963b285f6c2333576aaffb2c'
assert len(mapping['layers']) == 20
source_hashes = {}
transfers = []
unmatched = []
for candidate in mapping['mapped_candidates']:
    matches = [row for row in ledger['findings'] if (
        hashlib.sha256(('app/' + row['file']).encode()).hexdigest() == candidate['path_sha256']
        and row['line'] == candidate['line'] and row['rule'] == candidate['rule_id'])]
    if not matches:
        unmatched.append({'candidate_id': candidate['candidate_id'], 'reason': 'no_prior_locator'})
        continue
    assert len(matches) == 1
    previous = matches[0]
    file = previous['file']
    if file not in source_hashes:
        content = subprocess.run(['git', 'show', ledger['commit'] + ':' + file], cwd=ROOT, check=True, capture_output=True).stdout
        source_hashes[file] = hashlib.sha256(content).hexdigest()
    if source_hashes[file] != candidate['whole_file_sha256']:
        unmatched.append({'candidate_id': candidate['candidate_id'], 'reason': 'whole_file_bytes_differ', 'source_path': file, 'prior_sha256': source_hashes[file], 'image_sha256': candidate['whole_file_sha256']})
        continue
    transfers.append({'candidate_id': candidate['candidate_id'], 'source_path': file, 'line': candidate['line'], 'rule_id': candidate['rule_id'], 'classification': previous['classification'], 'source_commit': ledger['commit'], 'whole_file_sha256': source_hashes[file], 'image_blob': candidate['blob'], 'image_diff_id': candidate['diff_id']})
assert len(transfers) + len(unmatched) == 219
print(json.dumps({'mapping_sha256': hashlib.sha256(mapping_path.read_bytes()).hexdigest(), 'ledger_sha256': hashlib.sha256(LEDGER.read_bytes()).hexdigest(), 'source_hashes': source_hashes, 'transferred': transfers, 'remaining': unmatched, 'transferred_count': len(transfers), 'untriaged_count': len(unmatched)}, sort_keys=True))
